            const BASES = [
                { f: (x) => x * x, tex: (u) => `${u}^2`, key: [0, 0], name: 'вершина' },
                { f: (x) => (x >= 0 ? Math.sqrt(x) : NaN), tex: (u) => `\\sqrt{${u}}`, key: [0, 0], name: 'початок' },
                { f: (x) => (Math.abs(x) > 1e-6 ? 1 / x : NaN), tex: (u) => `\\dfrac{1}{${u}}`, key: null, name: '' },
            ];
            let base = 0, trShown = false;
            window.setBase = function (i) {
                base = i;
                [0, 1, 2].forEach(j => document.getElementById('base-' + j).classList.toggle('on', j === i));
                renderTr();
            };
            function trVals() {
                let k = parseFloat(document.getElementById('w-k').value);
                if (k === 0) { k = 0.5; document.getElementById('w-k').value = 0.5; }
                return [k, parseInt(document.getElementById('w-a').value, 10), parseInt(document.getElementById('w-b').value, 10)];
            }
            function trTex(k, a, b) {
                const inner = a === 0 ? 'x' : a > 0 ? `x - ${a}` : `x + ${-a}`;
                const B = BASES[base], core = B.tex(base === 0 && a !== 0 ? `(${inner})` : inner);
                const kk = k === 1 ? '' : k === -1 ? '-' : `${fmtN(k)}`.replace('−', '-');
                const bb = b === 0 ? '' : b > 0 ? ` + ${b}` : ` - ${-b}`;
                return `y = ${kk}${core}${bb}`;
            }
            window.renderTr = function () {
                const [k, a, b] = trVals();
                document.getElementById('w-k-val').textContent = fmtN(k);
                document.getElementById('w-a-val').textContent = fmtN(a);
                document.getElementById('w-b-val').textContent = fmtN(b);
                trShown = false;
                const f = document.getElementById('tr-formula');
                f.innerHTML = `$${trTex(k, a, b)}$`;
                typeset(f);
                document.getElementById('tr-text').innerHTML = '';
                drawTr(k, a, b);
            };
            function drawTr(k, a, b) {
                const B = BASES[base], g = (x) => k * B.f(x - a) + b;
                let svg = axesSvg(P0) + curveSvg(P0, B.f, -6, 6, () => 'ghost');
                if (base === 2) svg += curveSvg(P0, B.f, -6, -0.05, () => 'ghost').slice(0, 0);
                if (trShown) {
                    if (base === 2) svg += curveSvg(P0, g, -6, a - 0.02, () => 'pos') + curveSvg(P0, g, a + 0.02, 6, () => 'pos');
                    else svg += curveSvg(P0, g, -6, 6, () => 'pos');
                    if (B.key) svg += dot(P0, B.key[0] + a, k * B.key[1] + b, `(${fmtN(a)}; ${fmtN(b)})`);
                }
                document.getElementById('tr-scene').innerHTML = svg;
            }
            window.revealTr = function () {
                const [k, a, b] = trVals(), out = document.getElementById('tr-text');
