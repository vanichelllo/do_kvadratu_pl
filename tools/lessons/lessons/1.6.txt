            // --- Слайд 2: прямокутник із крісел ---
            let turned = false;
            const val = (id) => parseInt(document.getElementById(id).value, 10) || 1;
            window.renderSeats = function () {
                const r = val('w-r'), c = val('w-c');
                document.getElementById('w-r-val').textContent = r;
                document.getElementById('w-c-val').textContent = c;
                document.getElementById('seat-answer').classList.remove('shown');
                const rows = turned ? c : r, cols = turned ? r : c;
                const size = Math.min(36, 360 / cols, 270 / rows);
                const x0 = (400 - cols * size) / 2, y0 = (300 - rows * size) / 2;
                let svg = '';
                for (let i = 0; i < rows; i++) {
                    for (let j = 0; j < cols; j++) {
                        // Колір — за рядом початкового залу, щоб після повороту було видно, що ряди стали стовпчиками
                        const origRow = turned ? j : i;
                        svg += `<rect x="${x0 + j * size + 3}" y="${y0 + i * size + 3}" width="${size - 6}" height="${size - 6}" rx="${size / 5}" class="seat${origRow % 2 ? ' alt' : ''}"/>`;
                    }
                }
                document.getElementById('seat-scene').innerHTML = svg;
            };
            window.turnHall = function () { turned = !turned; renderSeats(); };
            window.revealSeats = function () {
                const r = val('w-r'), c = val('w-c');
                document.getElementById('seat-prod').textContent = turned ? `${r} · ${c} = ${r * c}` : `${c} · ${r} = ${r * c}`;
                document.getElementById('seat-why').textContent = `Рядів: ${r}, у кожному по ${c}. ${c} · ${r} = ${r} · ${c} = ${r * c} — повертай як хочеш, крісел стільки ж.`;
                document.getElementById('seat-answer').classList.add('shown');
            };

            // --- Слайд 4: множення в стовпчик на двоцифрове число ---
            const PAIRS = [[47, 36], [125, 24], [308, 15]];
            let pairIndex = 0, shownSteps = 0;
            function mulSteps(a, b) {
                const units = b % 10, tens = Math.floor(b / 10);
                return [
                    { text: `Множимо ${a} на одиниці: ${a} · ${units} = ${fmt(a * units)}.` },
                    { text: `Множимо ${a} на десятки: ${a} · ${tens} = ${fmt(a * tens)} — це десятки, тому пишемо зі зсувом на розряд вліво (${fmt(a * tens * 10)}).` },
                    { text: `Додаємо: ${fmt(a * units)} + ${fmt(a * tens * 10)} = ${fmt(a * b)}.` },
                    { text: `Відповідь: ${a} · ${b} = ${fmt(a * b)}.` },
                ];
            }
            // Числа в клітинках, вирівняні праворуч; shift — зсув вліво на кількість розрядів
            function rowCells(str, y, n, w, x0, cls, shift) {
                let svg = '';
                str.split('').forEach((d, k) => {
                    const i = n - 1 - shift - (str.length - 1 - k);
                    svg += `<text x="${x0 + i * w + w / 2}" y="${y + 26}" class="${cls}">${d}</text>`;
                });
                return svg;
            }
            window.renderColumn = function () {
                const [a, b] = PAIRS[pairIndex];
                const n = String(a * b).length + 1, w = 38, x0 = 300 - n * w;
                const p1 = String(a * (b % 10)), p2 = String(a * Math.floor(b / 10));
                let svg = '';
                for (let i = 0; i < n; i++) {
                    svg += `<rect x="${x0 + i * w + 2}" y="10" width="${w - 4}" height="36" rx="5" class="col-cell"/>`
                         + `<rect x="${x0 + i * w + 2}" y="52" width="${w - 4}" height="36" rx="5" class="col-cell"/>`;
                }
                svg += rowCells(String(a), 10, n, w, x0, 'col-digit', 0) + rowCells(String(b), 52, n, w, x0, 'col-digit', 0)
                     + `<text x="${x0 - 14}" y="80" class="col-sign">×</text>`
                     + `<line x1="${x0}" y1="98" x2="${x0 + n * w}" y2="98" class="col-line"/>`;
                if (shownSteps >= 1) svg += rowCells(p1, 104, n, w, x0, 'col-part', 0);
                if (shownSteps >= 2) svg += rowCells(p2, 144, n, w, x0, 'col-part', 1)
                     + `<text x="${x0 + (n - 1) * w + 6}" y="170" class="col-shift">зсув</text>`
                     + `<text x="${x0 - 14}" y="166" class="col-sign">+</text>`;
                if (shownSteps >= 3) svg += `<line x1="${x0}" y1="186" x2="${x0 + n * w}" y2="186" class="col-line"/>`
                     + rowCells(String(a * b), 192, n, w, x0, 'col-result', 0);
                document.getElementById('col-scene').innerHTML = svg;
                const box = document.getElementById('col-steps');
                box.innerHTML = '';
                mulSteps(a, b).slice(0, shownSteps).forEach(s => { const div = document.createElement('div'); div.textContent = s.text; box.appendChild(div); });
                if (!shownSteps) box.textContent = `Скільки буде ${a} · ${b}? Спершу прикиньте, а потім показуємо кроки.`;
            };
            window.colStep = function () { shownSteps = Math.min(shownSteps + 1, 4); renderColumn(); };
            window.colReset = function () { shownSteps = 0; renderColumn(); };
            window.colPair = function () { pairIndex = (pairIndex + 1) % PAIRS.length; shownSteps = 0; renderColumn(); };

            // --- Слайд 5: 34 · 23 без зсуву і зі зсувом ---
            function trapColumn(shifted) {
                const n = 4, w = 34, x0 = 40;
                let svg = '';
                for (let i = 0; i < n; i++) {

                }
                svg += rowCells('34', 8, n, w, x0, 'col-digit', 0) + rowCells('23', 48, n, w, x0, 'col-digit', 0)
                     + `<text x="${x0 - 14}" y="76" class="col-sign">×</text><line x1="${x0}" y1="94" x2="${x0 + n * w}" y2="94" class="col-line"/>`
                     + rowCells('102', 98, n, w, x0, 'col-part', 0) + rowCells('68', 136, n, w, x0, 'col-part', shifted ? 1 : 0)
