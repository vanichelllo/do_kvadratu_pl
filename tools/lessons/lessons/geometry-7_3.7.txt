            const f1 = (v) => v.toFixed(1);
            const ln = (p, q, cls) => `<line x1="${f1(p[0])}" y1="${f1(p[1])}" x2="${f1(q[0])}" y2="${f1(q[1])}" class="${cls}"/>`;
            const poly = (P) => `<polygon points="${P.map(p => p.map(f1).join(',')).join(' ')}" class="tri"/>`;
            const nm = (p, s, dx, dy) => `<text x="${f1(p[0] + dx)}" y="${f1(p[1] + dy)}" class="v-name">${s}</text>`;
            function segs(A, B, C, show = [true, true, true]) {
                const M = [(A[0] + C[0]) / 2, (A[1] + C[1]) / 2];
                const a = Math.hypot(B[0] - C[0], B[1] - C[1]), c = Math.hypot(B[0] - A[0], B[1] - A[1]);
                const L = [A[0] + (C[0] - A[0]) * c / (a + c), A[1] + (C[1] - A[1]) * c / (a + c)];
                const H = [B[0], A[1]];
                let s = '';
                if (H[0] < A[0]) s += ln(A, [H[0] - 10, H[1]], 'ext');
                if (H[0] > C[0]) s += ln(C, [H[0] + 10, H[1]], 'ext');
                if (show[2]) s += ln(B, H, 'alt') + `<path d="M${f1(H[0] + (H[0] <= C[0] ? 12 : -12))} ${H[1]} v-12 h${H[0] <= C[0] ? -12 : 12}" class="rm"/>`;
                if (show[1]) s += ln(B, L, 'bis');
                if (show[0]) s += ln(B, M, 'med');
                return s;
            }
