            // Піца з n рівних шматків, перші m зафарбовано
            function pizzaSvg(cx, cy, r, n, m) {
                let svg = '';
                for (let k = 0; k < n; k++) {
                    const a0 = -Math.PI / 2 + 2 * Math.PI * k / n, a1 = -Math.PI / 2 + 2 * Math.PI * (k + 1) / n;
                    const x0 = cx + r * Math.cos(a0), y0 = cy + r * Math.sin(a0), x1 = cx + r * Math.cos(a1), y1 = cy + r * Math.sin(a1);
                    const big = a1 - a0 > Math.PI ? 1 : 0;
                    svg += `<path d="M${cx} ${cy} L${x0.toFixed(1)} ${y0.toFixed(1)} A${r} ${r} 0 ${big} 1 ${x1.toFixed(1)} ${y1.toFixed(1)} Z" class="slice${k < m ? ' on' : ''}"/>`;
                }
                return svg + `<circle cx="${cx}" cy="${cy}" r="${r}" class="crust"/>`;
            }
            // Дріб, намальований у SVG: чисельник, риска, знаменник
            function fracSvg(x, y, m, n, captions) {
                let svg = `<text x="${x}" y="${y - 12}" class="frac-num">${m}</text><line x1="${x - 34}" y1="${y}" x2="${x + 34}" y2="${y}" class="frac-bar"/><text x="${x}" y="${y + 46}" class="frac-den">${n}</text>`;
                if (captions) svg += `<text x="${x + 44}" y="${y - 24}" class="frac-cap" style="fill:var(--brand-blue)">чисельник</text><text x="${x + 44}" y="${y + 34}" class="frac-cap" style="fill:var(--brand-green)">знаменник</text>`;
                return svg;
            }

            // Смужка з n частин, перші m зафарбовано
            function stripSvg(x, y, w, h, n, m, cls) {
                let svg = '';
                for (let k = 0; k < n; k++) svg += `<rect x="${(x + k * w / n).toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="strip-cell"/>`;
                for (let k = 0; k < Math.min(m, n); k++) svg += `<rect x="${(x + k * w / n).toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="${cls}" style="stroke:var(--text-dark);stroke-width:2"/>`;
                return svg;
            }

            function stripVals() {
                const n = parseInt(document.getElementById('w-n').value, 10) || 2;
                ['w-a', 'w-b'].forEach(id => { const s = document.getElementById(id); s.max = n; if (parseInt(s.value, 10) > n) s.value = n; });
                return [parseInt(document.getElementById('w-a').value, 10) || 0, parseInt(document.getElementById('w-b').value, 10) || 0, n];
            }
            function stripsScene(a, b, n, sign) {
                return stripSvg(20, 30, 300, 50, n, a, 'strip-a') + stripSvg(20, 130, 300, 50, n, b, 'strip-b')

            }

                    : `${a}/${n} ${sign} ${b}/${n}, бо ${a} ${sign} ${b}`;
                document.getElementById('strips-answer').classList.add('shown');
