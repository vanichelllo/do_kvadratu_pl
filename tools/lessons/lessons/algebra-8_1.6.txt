            // Нескоротний дріб p/q у TeX (знак — перед дробом)
            function fracTex(p, q) {
                if (q < 0) { p = -p; q = -q; }
                const g = gcd(Math.abs(p), q) || 1;
                p /= g; q /= g;
                if (q === 1) return String(p);
                return (p < 0 ? '-' : '') + `\\dfrac{${Math.abs(p)}}{${q}}`;
            }
