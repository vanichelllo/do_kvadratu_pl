            // --- Слайд 2: на скільки A більше за B ---
            const RAY_X0 = 25, RAY_DX = 19;
            const val = (id) => parseInt(document.getElementById(id).value, 10) || 0;
            window.renderGap = function () {
                const a = Math.max(1, val('w-a'));
                const bInput = document.getElementById('w-b');
                bInput.max = a;                                   // від'ємник не більший за зменшуване
                if (val('w-b') > a) bInput.value = a;
                const b = val('w-b');
                document.getElementById('w-a-val').textContent = a;
                document.getElementById('w-b-val').textContent = b;
                document.getElementById('gap-answer').classList.remove('shown');
                let svg = `<line x1="15" y1="90" x2="425" y2="90" class="ray"/><path d="M430 90 l-10 -6 v12 z" class="ray-arrow"/>`;
                for (let v = 0; v <= 20; v++) {
                    const x = RAY_X0 + v * RAY_DX;
                    svg += `<line x1="${x}" y1="84" x2="${x}" y2="96" class="ray-tick"/><text x="${x}" y="114" class="ray-label">${v}</text>`;
                }
                const xa = RAY_X0 + a * RAY_DX, xb = RAY_X0 + b * RAY_DX;
                if (a > b) svg += `<line x1="${xb}" y1="55" x2="${xa}" y2="55" class="gap-line"/><line x1="${xb}" y1="48" x2="${xb}" y2="62" class="gap-line"/><line x1="${xa}" y1="48" x2="${xa}" y2="62" class="gap-line"/><text x="${(xa + xb) / 2}" y="44" class="gap-cap">?</text>`;
                svg += `<circle cx="${xb}" cy="90" r="8" class="pt-b"/><text x="${xb}" y="134" class="pt-cap pt-b">B</text>`
                     + `<circle cx="${xa}" cy="90" r="${a === b ? 5 : 8}" class="pt-a"/><text x="${a === b ? xa + 14 : xa}" y="134" class="pt-cap pt-a">A</text>`;
                document.getElementById('gap-scene').innerHTML = svg;
            };
            window.revealGap = function () {
                const a = val('w-a'), b = val('w-b');
                document.getElementById('gap-sign').textContent = `${a} − ${b} = ${a - b}`;
                document.getElementById('gap-check').textContent = a === b
                    ? 'Точки збігаються: різниця 0.'
                    : `Перевірка: ${a - b} + ${b} = ${a} ✓`;
                const cap = document.querySelector('#gap-scene .gap-cap');
                if (cap) cap.textContent = a - b;
                document.getElementById('gap-answer').classList.add('shown');
            };

            // --- Слайд 4: віднімання в стовпчик із займанням ---
            const PLACES = ['Одиниці', 'Десятки', 'Сотні', 'Тисячі', 'Десятки тисяч'];
            const PLACES_GEN = ['одиниць', 'десятків', 'сотень', 'тисяч', 'десятків тисяч'];
            const PAIRS = [[731, 258], [503, 128], [1000, 1]];
            let pairIndex = 0, shownSteps = 0;
            function columnSteps(a, b) {
                // Кожен крок: розряд, цифра відповіді, верхні цифри після займання, пояснення
                const top = String(a).split('').reverse().map(Number), bot = String(b).split('').reverse().map(Number);
                const steps = [], resultLen = String(a - b).length;
                for (let i = 0; i < top.length; i++) {
                    const y = bot[i] || 0;
                    let text = `${PLACES[i]}: `;
                    if (top[i] < y) {
                        let j = i + 1;
                        while (top[j] === 0) j++;
                        text += `${top[i]} менше за ${y} — займаємо 1 у розряді ${PLACES_GEN[j]}`
                              + (j > i + 1 ? ', а нулі між ними стають 9' : '') + '. ';
                        top[j] -= 1;
                        for (let k = j - 1; k > i; k--) top[k] = 9;
                        top[i] += 10;
                    }
                    const d = top[i] - y;
                    if (i >= resultLen) text += `${top[i]} − ${y} = 0 — нуль на початку числа не пишемо.`;
                    else text += `${top[i]} − ${y} = ${d}.`;
                    steps.push({ col: i, digit: i >= resultLen ? null : d, tops: top.slice(), text });
                }
                steps.push({ col: -1, tops: top.slice(), text: `Відповідь: ${fmt(a)} − ${fmt(b)} = ${fmt(a - b)}. Перевірка: ${fmt(a - b)} + ${fmt(b)} = ${fmt(a)} ✓` });
                return steps;
            }
            window.renderColumn = function () {
                const [a, b] = PAIRS[pairIndex];
                const steps = columnSteps(a, b), shown = steps.slice(0, shownSteps);
                const n = String(a).length, w = 52, x0 = 60;
                const xOf = (i) => x0 + (n - 1 - i) * w;
                const orig = String(a).split('').reverse().map(Number), sb = String(b).split('').reverse();
                const last = shown[shown.length - 1];
                const tops = last ? last.tops : orig;
                const active = last && last.col >= 0 ? last.col : -1;
                let svg = '';
                for (let i = 0; i < n; i++) {
                    const x = xOf(i), changed = tops[i] !== orig[i];
                    svg += `<rect x="${x + 3}" y="40" width="${w - 6}" height="44" rx="6" class="col-cell${i === active ? ' active' : ''}"/>`
                         + `<text x="${x + w / 2}" y="72" class="col-digit${changed ? ' borrowed' : ''}">${orig[i]}</text>`
                         + (changed ? `<text x="${x + w / 2}" y="30" class="col-new">${tops[i]}</text>` : '')
                         + `<rect x="${x + 3}" y="90" width="${w - 6}" height="44" rx="6" class="col-cell${i === active ? ' active' : ''}"/>`
                         + `<text x="${x + w / 2}" y="122" class="col-digit">${sb[i] || ''}</text>`;
                }
                svg += `<text x="${xOf(n - 1) - 16}" y="120" class="col-sign">−</text>`
                     + `<line x1="${xOf(n - 1) - 4}" y1="145" x2="${xOf(0) + w}" y2="145" class="col-line"/>`;
                shown.forEach(s => {
                    if (s.col >= 0 && s.digit !== null) svg += `<text x="${xOf(s.col) + w / 2}" y="182" class="col-result">${s.digit}</text>`;
                });
                document.getElementById('col-scene').innerHTML = svg;
                const box = document.getElementById('col-steps');
                box.innerHTML = '';
                shown.forEach(s => { const div = document.createElement('div'); div.textContent = s.text; box.appendChild(div); });
                if (!shown.length) box.textContent = `Скільки буде ${fmt(a)} − ${fmt(b)}? Скажіть, а потім показуємо кроки.`;
            };
            window.colStep = function () {
                const [a, b] = PAIRS[pairIndex];
                shownSteps = Math.min(shownSteps + 1, columnSteps(a, b).length);
                renderColumn();
            };
            window.colReset = function () { shownSteps = 0; renderColumn(); };
            window.colPair = function () { pairIndex = (pairIndex + 1) % PAIRS.length; shownSteps = 0; renderColumn(); };

            // --- Слайд 5: неправильне і правильне віднімання 52 − 17 ---
            function stack(result, wrong) {
                const w = 44, x0 = 70;
                const rows = [['5', '2'], ['1', '7'], result.split('')];
                let svg = '';
                rows.forEach((row, r) => row.forEach((d, i) => {
                    const y = [30, 80, 156][r];
                    if (r < 2) svg += `<rect x="${x0 + i * w + 2}" y="${y}" width="${w - 4}" height="42" rx="5" class="col-cell"/>`;
                    svg += `<text x="${x0 + i * w + w / 2}" y="${y + 31}" class="${r === 2 ? 'col-result' : 'col-digit'}"${r === 2 && wrong ? ' style="fill:var(--warning-red)"' : ''}>${d}</text>`;
                }));
                svg += `<text x="${x0 - 14}" y="110" class="col-sign">−</text><line x1="${x0}" y1="134" x2="${x0 + 2 * w}" y2="134" class="col-line"/>`;
                if (wrong) svg += `<line x1="${x0 - 6}" y1="150" x2="${x0 + 2 * w + 6}" y2="196" class="cross"/>`;
                else svg += `<text x="${x0 + w * 1.5}" y="24" class="col-new">12</text><text x="${x0 + w / 2}" y="24" class="col-new">4</text>`;
                return svg;
            }
            window.renderTrap = function () {
                document.getElementById('trap-wrong').innerHTML = stack('45', true);
                document.getElementById('trap-right').innerHTML = stack('35', false);
            };
