            // Підпис точки: трохи назовні від центру c
            function vName(p, c, name) {
                const dx = p[0] - c[0], dy = p[1] - c[1], L = Math.hypot(dx, dy) || 1;
                return `<text x="${f1(p[0] + dx / L * 18)}" y="${f1(p[1] + dy / L * 18 + 6)}" class="v-name">${name}</text>`;
            }
            // n рисок у середині відрізка pq (позначка рівних частин)
            function ticks(p, q, n) {
                const m = mid(p, q), dx = q[0] - p[0], dy = q[1] - p[1], L = Math.hypot(dx, dy), ux = dx / L, uy = dy / L;
                let s = '';
                for (let k = 0; k < n; k++) {
                    const o = (k - (n - 1) / 2) * 6, c = [m[0] + ux * o, m[1] + uy * o];
                    s += seg([c[0] - uy * 7, c[1] + ux * 7], [c[0] + uy * 7, c[1] - ux * 7], 'tick-mark');
                }
                return s;
            }
            // Трикутник ABC (A, C — основа) з середньою лінією MN: M — середина AB, N — середина BC
            function triWithMid(A, B, C, opts = {}) {
                const M = mid(A, B), N = mid(B, C), c = [(A[0] + B[0] + C[0]) / 3, (A[1] + B[1] + C[1]) / 3];

            function triScene() {
                const t = triPts(), T = triWithMid(t.A, t.B, t.C);
                let s = '';
                if (allMid) {
                    const K = mid(t.A, t.C);
                    s += poly([T.M, T.N, K], 'part p2') + poly([t.A, T.M, K], 'part p1') + poly([K, T.N, t.C], 'part p1') + poly([T.M, t.B, T.N], 'part p1');
                    s += seg(T.M, K, 'mid faint') + seg(T.N, K, 'mid faint') + dot(K);
                }
                s += T.svg;
                if (triShown) s += `<text x="${f1((T.M[0] + T.N[0]) / 2)}" y="${f1(T.M[1] - 10)}" class="svg-caption g">${num(t.a / 2)}</text><text x="${f1((t.A[0] + t.C[0]) / 2)}" y="${Y0 + 24}" class="svg-caption">${t.a}</text>`;
