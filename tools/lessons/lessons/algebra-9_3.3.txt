            window.renderLab = function () {
                const q = +document.getElementById('w-q').value;
                document.getElementById('w-q-val').textContent = String(q).replace('.', ',');
                if (q === 0) { document.getElementById('lab-seq').innerHTML = '$q = 0$ — не прогресія!'; document.getElementById('lab-scene').innerHTML = ''; return; }
                const seq = Array.from({ length: 7 }, (_, k) => Math.pow(q, k));
                const mx = Math.max(...seq.map(Math.abs)), h = 100 / mx, y0 = 130;
                let s = `<line x1="10" y1="${y0}" x2="390" y2="${y0}" stroke="currentColor" stroke-width="1.5"/>`;
                seq.forEach((v, k) => {
                    const x = 20 + k * 53, top = v >= 0 ? y0 - v * h : y0;
                    s += `<rect x="${x}" y="${top}" width="40" height="${Math.max(Math.abs(v) * h, 1)}" class="bar${v < 0 ? ' neg' : ''}"/>`;
                    s += `<text x="${x + 20}" y="${v >= 0 ? top - 6 : top + Math.abs(v) * h + 16}" class="svg-caption">${String(Math.round(v * 1000) / 1000).replace('.', ',').replace('-', '−')}</text>`;
                });
                document.getElementById('lab-scene').innerHTML = s;
                const t = document.getElementById('lab-seq');
                t.innerHTML = `$1;\\; ${String(q).replace('.', '{,}')};\\; ${String(q * q).replace('.', '{,}')};\\; \\ldots$`;
                document.getElementById('lab-out').classList.remove('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([t]);
            };
            window.revealLab = function () {
                const q = +document.getElementById('w-q').value, o = document.getElementById('lab-out');
                o.innerHTML = q > 1 ? 'Зростає (при $b_1 \\gt 0$)' : q === 1 ? 'Стала' : q > 0 ? 'Спадає, наближаючись до $0$' : 'Знаки чергуються — не монотонна';
                o.classList.add('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([o]);
            };
