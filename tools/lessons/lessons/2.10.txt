            // Коса проекція: x — довжина (вправо), y — ширина (углиб, угору-праворуч), z — висота (вгору)
            const K = 0.45;
            const pt = (x0, y0, s, x, y, z) => [x0 + x * s + y * s * K, y0 - z * s - y * s * K];
            const pts = (arr) => arr.map(p => p[0].toFixed(1) + ',' + p[1].toFixed(1)).join(' ');
            function cubeSvg(x0, y0, s, i, j, k) {
                const P = (x, y, z) => pt(x0, y0, s, i + x, j + y, k + z);
                return `<polygon points="${pts([P(0, 0, 0), P(1, 0, 0), P(1, 0, 1), P(0, 0, 1)])}" class="cf"/>`
                    + `<polygon points="${pts([P(0, 0, 1), P(1, 0, 1), P(1, 1, 1), P(0, 1, 1)])}" class="ct"/>`
                    + `<polygon points="${pts([P(1, 0, 0), P(1, 1, 0), P(1, 1, 1), P(1, 0, 1)])}" class="cr"/>`;
            }
            // Каркас коробки a × b × c (невидимі ребра — пунктир); water — заливка передньої грані до висоти w
            function frameSvg(x0, y0, s, a, b, c, water) {
                const P = (x, y, z) => pt(x0, y0, s, x, y, z);
                const L = (p, q, h) => `<line x1="${p[0].toFixed(1)}" y1="${p[1].toFixed(1)}" x2="${q[0].toFixed(1)}" y2="${q[1].toFixed(1)}" class="edge${h ? ' hidden' : ''}"/>`;
                let svg = `<polygon points="${pts([P(0, 0, 0), P(a, 0, 0), P(a, 0, c), P(0, 0, c)])}" class="face"/>`;
                if (water) svg += `<polygon points="${pts([P(0, 0, 0), P(a, 0, 0), P(a, 0, water), P(0, 0, water)])}" class="face water"/>`
                    + `<polygon points="${pts([P(a, 0, 0), P(a, b, 0), P(a, b, water), P(a, 0, water)])}" class="face water"/>`;
                svg += L(P(0, b, 0), P(a, b, 0), 1) + L(P(0, b, 0), P(0, b, c), 1) + L(P(0, 0, 0), P(0, b, 0), 1);
                [[P(0, 0, 0), P(a, 0, 0)], [P(a, 0, 0), P(a, 0, c)], [P(a, 0, c), P(0, 0, c)], [P(0, 0, c), P(0, 0, 0)],
                 [P(a, 0, 0), P(a, b, 0)], [P(a, 0, c), P(a, b, c)], [P(0, 0, c), P(0, b, c)], [P(a, b, 0), P(a, b, c)], [P(0, b, c), P(a, b, c)]]
                    .forEach(([p, q]) => { svg += L(p, q, 0); });
                return svg;
            }

            function cubeVals() { return ['w-a', 'w-b', 'w-c'].map(id => parseInt(document.getElementById(id).value, 10) || 1); }
            function cubesScene(a, b, c, n) {
                // Кубики додаємо шарами знизу вгору; малюємо від дальніх до ближніх, щоб ближні перекривали
                const list = [];
                for (let k = 0; k < c; k++) for (let j = 0; j < b; j++) for (let i = 0; i < a; i++) list.push([i, j, k]);
                const shown = list.slice(0, n).sort((p, q) => (q[1] - p[1]) || (p[2] - q[2]) || (p[0] - q[0]));
                let svg = frameSvg(X2, Y2, S2, a, b, c, 0);
                shown.forEach(([i, j, k]) => { svg += cubeSvg(X2, Y2, S2, i, j, k); });
                return svg + `<text x="${X2 + a * S2 / 2}" y="${Y2 + 18}" class="len">${a} см</text>`;
            }
            function cubesText(a, b, c, n) {
                if (n === 0) return '';
                const layer = a * b;
                if (n < a * b * c) return `Шар: ${a} · ${b} = ${layer} кубиків. Кубиків: ${n}`;
                return `${c} шар(и) по ${layer}: ${a} · ${b} · ${c} = ${a * b * c} см³`.replace('шар(и)', c === 1 ? 'шар' : 'шари');
            }
            function showCubes(a, b, c, n) {
                document.getElementById('cube-scene').innerHTML = cubesScene(a, b, c, n);
                const el = document.getElementById('fill-count');
                el.textContent = cubesText(a, b, c, n);
                el.classList.toggle('done', n === a * b * c);
            }
            window.renderCubes = function () {
                clearInterval(fillTimer);
                const [a, b, c] = cubeVals();
                document.getElementById('w-a-val').textContent = a + ' см';
                document.getElementById('w-b-val').textContent = b + ' см';
                document.getElementById('w-c-val').textContent = c + ' см';
                showCubes(a, b, c, 0);
            };
            window.fillCubes = function () {
                renderCubes();
                const [a, b, c] = cubeVals(), total = a * b * c;
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) { showCubes(a, b, c, total); return; }
                let n = 0;
                fillTimer = setInterval(() => {
                    n = Math.min(total, n + a);   // по ряду кубиків за раз
                    showCubes(a, b, c, n);
                    if (n >= total) clearInterval(fillTimer);
                }, Math.max(150, Math.round(3000 / (b * c))));
            };
