            // --- Транспортир з центром (cx, cy) і радіусом r; кут відкладаємо від правої сторони ---
            const rad = (d) => d * Math.PI / 180;
            const pol = (cx, cy, r, d) => ({ x: cx + r * Math.cos(rad(d)), y: cy - r * Math.sin(rad(d)) });
            function protractor(cx, cy, r, inner) {
                let svg = `<path d="M${cx - r} ${cy} A${r} ${r} 0 0 1 ${cx + r} ${cy} Z" class="prot"/>`;
                for (let d = 0; d <= 180; d += 5) {
                    const big = d % 10 === 0, a = pol(cx, cy, r, d), b = pol(cx, cy, r - (big ? 12 : 7), d);
                    svg += `<line x1="${a.x}" y1="${a.y}" x2="${b.x}" y2="${b.y}" class="prot-tick${big ? ' big' : ''}"/>`;
                    if (d % 10 === 0) {
                        const t = pol(cx, cy, r - 22, d);
                        svg += `<text x="${t.x}" y="${t.y + 4}" class="prot-num">${d}</text>`;
                        if (inner) { const u = pol(cx, cy, r - 38, d); svg += `<text x="${u.x}" y="${u.y + 4}" class="prot-num inner">${180 - d}</text>`; }
                    }
                }
                return svg;
            }
            function angleSvg(cx, cy, len, deg) {
                const b = pol(cx, cy, len, deg), cls = deg < 90 ? '' : deg === 90 ? ' right' : ' obtuse';
                let arc;
                if (deg === 90) {
                    arc = `<path d="M${cx + 22} ${cy} L${cx + 22} ${cy - 22} L${cx} ${cy - 22}" class="angle-arc right"/>`;
                } else {
                    const r = 34, p = pol(cx, cy, r, deg);
                    arc = `<path d="M${cx} ${cy} L${cx + r} ${cy} A${r} ${r} 0 ${deg > 180 ? 1 : 0} 0 ${p.x} ${p.y} Z" class="angle-arc${cls}"/>`;
                }
                return arc + `<line x1="${cx}" y1="${cy}" x2="${cx + len}" y2="${cy}" class="side"/>`
                     + `<line x1="${cx}" y1="${cy}" x2="${b.x}" y2="${b.y}" class="side"/>`
                     + `<circle cx="${cx}" cy="${cy}" r="5" class="pt"/>`
                     + `<text x="${cx}" y="${cy + 22}" class="pt-name">O</text>`
                     + `<text x="${cx + len + 4}" y="${cy + 22}" class="pt-name">A</text>`
                     + `<text x="${pol(cx, cy, len + 16, deg).x}" y="${pol(cx, cy, len + 16, deg).y + 6}" class="pt-name">B</text>`;
            }
            function kind(d) { return d < 90 ? 'гострий' : d === 90 ? 'прямий' : d < 180 ? 'тупий' : 'розгорнутий'; }

            // --- Слайд 1: годинники 3:00 і 6:00 ---
            function clock(cx, cy, h, label) {
                let svg = `<circle cx="${cx}" cy="${cy}" r="70" class="clock-face"/>`;
                for (let k = 0; k < 12; k++) {
                    const a = pol(cx, cy, 70, 90 - k * 30), b = pol(cx, cy, 60, 90 - k * 30);
                    svg += `<line x1="${a.x}" y1="${a.y}" x2="${b.x}" y2="${b.y}" class="prot-tick big"/>`;
                }
                const m = pol(cx, cy, 55, 90), hh = pol(cx, cy, 40, 90 - h * 30);
                svg += `<line x1="${cx}" y1="${cy}" x2="${m.x}" y2="${m.y}" class="hand min"/>`
                     + `<line x1="${cx}" y1="${cy}" x2="${hh.x}" y2="${hh.y}" class="hand"/>`
                     + `<text x="${cx}" y="${cy + 95}" class="svg-caption">${label}</text>`;
                return svg;
            }
