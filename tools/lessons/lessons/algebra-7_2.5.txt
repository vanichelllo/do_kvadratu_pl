            const typeset = (el) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([el]); };

            // --- Слайд 1: город 3a × (a + 2b + 4) ---
            let gardenShown = false;
            window.renderGarden = function () {
                const x0 = 50, y0 = 40, h = 150, ws = [100, 150, 90], labels = ['a', '2b', '4'], areas = ['3a²', '6ab', '12a'];
                let s = '', x = x0;
                ws.forEach((w, i) => {
                    s += `<rect x="${x}" y="${y0}" width="${w}" height="${h}" class="part p${i + 1}"/>`
                        + `<text x="${x + w / 2}" y="${y0 - 10}" class="side">${labels[i]}</text>`
                        + `<text x="${x + w / 2}" y="${y0 + h / 2 + 7}" class="area${gardenShown ? '' : ' q'}">${gardenShown ? areas[i] : '?'}</text>`;
                    x += w;
                });
                s += `<text x="${x0 - 22}" y="${y0 + h / 2 + 6}" class="side">3a</text>`;
                s += `<text x="${x0 + 170}" y="${y0 + h + 32}" class="area${gardenShown ? '' : ' q'}">${gardenShown ? 'S = 3a² + 6ab + 12a' : 'S = ?'}</text>`;
                document.getElementById('garden-scene').innerHTML = s;
            };
            window.revealGarden = function () { gardenShown = true; renderGarden(); document.getElementById('answer-1').classList.add('shown'); };

            // --- Слайд 2: клік по «?» відкриває відповідь ---
            window.showCell = function (td) {
                const v = td.querySelector('.hidden-answer');
                if (v) v.classList.add('shown');
            };

            // --- Слайд 4: kx(mx + n) ---
            const coef = (c, v) => (c === 1 ? '' : c) + v;   // 1x → x
            function areaVals() { return ['w-k', 'w-m', 'w-n'].map(id => parseInt(document.getElementById(id).value, 10)); }
            let areaShown = false;
            function drawArea() {
                const [k, m, n] = areaVals(), X = 55, U = 11, x0 = 40, y0 = 40;
                const w1 = m * X, w2 = n * U * 2, h = k * 40;      // схематично: x ≈ 55 px, 1 ≈ 22 px по довжині
                let s = `<rect x="${x0}" y="${y0}" width="${w1}" height="${h}" class="part p1"/><rect x="${x0 + w1}" y="${y0}" width="${w2}" height="${h}" class="part p2"/>`;
                s += `<text x="${x0 + w1 / 2}" y="${y0 - 10}" class="side">${coef(m, 'x')}</text><text x="${x0 + w1 + w2 / 2}" y="${y0 - 10}" class="side">${n}</text>`;
                s += `<text x="${x0 - 8}" y="${y0 + h / 2 + 6}" class="side" style="text-anchor:end">${coef(k, 'x')}</text>`;
                s += `<text x="${x0 + w1 / 2}" y="${y0 + h / 2 + 7}" class="area${areaShown ? '' : ' q'}">${areaShown ? coef(k * m, 'x²') : '?'}</text>`;
                s += `<text x="${x0 + w1 + w2 / 2}" y="${y0 + h / 2 + 7}" class="area${areaShown ? '' : ' q'}">${areaShown ? k * n + 'x' : '?'}</text>`;
                document.getElementById('area-scene').innerHTML = s;
            }
            window.renderArea = function () {
                const [k, m, n] = areaVals();
                document.getElementById('w-k-val').textContent = k;
                document.getElementById('w-m-val').textContent = m;
                document.getElementById('w-n-val').textContent = n;
                areaShown = false;
                const t = document.getElementById('area-task');
                t.innerHTML = `$S = ${coef(k, 'x')}(${coef(m, 'x')} + ${n}) = \\,?$`;
                document.getElementById('area-result').innerHTML = '';
                typeset(t);
                drawArea();
            };
