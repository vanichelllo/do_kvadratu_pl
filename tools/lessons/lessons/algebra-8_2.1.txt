            window.renderPar = function () {
                const P = plane(-4, 4, -1, 10, 420, 300);
                let s = axesSvg(P, 1, 1) + curveSvg(P, (x) => x * x, -3.2, 3.2);
                [-3, -2, -1, 0, 1, 2, 3].forEach(x => s += `<circle cx="${P.sx(x)}" cy="${P.sy(x * x)}" r="5" class="pt"/>`);
                document.getElementById('par-scene').innerHTML = s;
            };
            window.renderLab = function () {
                const a = +document.getElementById('w-a').value, P = plane(-4, 4, -4, 10, 420, 300);
                document.getElementById('w-a-val').textContent = String(a).replace('.', ',');
                let s = axesSvg(P, 1, 2) + curveSvg(P, (x) => x * x, -3.2, 3.2);
                s += `<line x1="${P.sx(-4)}" y1="${P.sy(a)}" x2="${P.sx(4)}" y2="${P.sy(a)}" class="hline"/>`;
                if (a >= 0) { const r = Math.sqrt(a); s += `<circle cx="${P.sx(r)}" cy="${P.sy(a)}" r="5" class="pt"/><circle cx="${P.sx(-r)}" cy="${P.sy(a)}" r="5" class="pt"/>`; }
                document.getElementById('lab-scene').innerHTML = s;
                document.getElementById('lab-out').classList.remove('shown');
            };
            window.revealLab = function () {
                const a = +document.getElementById('w-a').value, o = document.getElementById('lab-out');
                o.innerHTML = a > 0 ? `$a \\gt 0$: два розв'язки, протилежні числа` : a === 0 ? `$a = 0$: один розв'язок $x = 0$` : `$a \\lt 0$: розв'язків немає`;
                o.classList.add('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([o]);
            };
