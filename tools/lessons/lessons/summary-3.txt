            };

            // --- Слайд 5: змішані задачі покроково ---
            const TASKS = [
                { task: 'Торт розрізали на 12 рівних шматків. Оля з\'їла 1/4 торта, Тарас — 5/12 торта. Яка частина торта лишилася?',
                  lines: ['1/4 від 12 шматків: 12 : 4 = 3 шматки, тобто 3/12 торта.', 'Разом з\'їли: 3/12 + 5/12 = 8/12.', 'Увесь торт — 12/12.', 'Лишилося: 12/12 − 8/12 = 4/12.', 'Відповідь: 4/12 торта (4 шматки).'] },
                { task: 'Першого дня велосипедист проїхав 3/8 маршруту — 24 км, а другого — 2/5 решти. Скільки кілометрів йому лишилося?',
                  lines: ['Відома частина — шукаємо число за дробом: 24 : 3 · 8 = 64 км — увесь маршрут.', 'Решта після першого дня: 64 − 24 = 40 км.', 'Другого дня — дріб від числа: 40 : 5 · 2 = 16 км.', 'Лишилося: 40 − 16 = 24 км.', 'Відповідь: 24 км.'] },
            ];
            let taskIndex = 0, shownSol = 0;
            window.renderSolution = function () {
                const T = TASKS[taskIndex];
                document.getElementById('sol-task').textContent = T.task;
                const box = document.getElementById('sol-lines');
                box.innerHTML = '';
                T.lines.slice(0, shownSol).forEach(l => { const d = document.createElement('div'); d.className = 'sol-line'; d.textContent = l; box.appendChild(d); });
            };
            window.solStep = function () { shownSol = Math.min(shownSol + 1, TASKS[taskIndex].lines.length); renderSolution(); };
            window.solReset = function () { shownSol = 0; renderSolution(); };
            window.solNext = function () { taskIndex = (taskIndex + 1) % TASKS.length; shownSol = 0; renderSolution(); };
