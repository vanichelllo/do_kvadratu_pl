            // Прямокутник a × b клітинок з лініями сітки всередині
            function cellsRect(x0, y0, a, b, s, cls) {
                let svg = `<rect x="${x0}" y="${y0}" width="${a * s}" height="${b * s}" class="${cls}"/>`;
                for (let i = 1; i < a; i++) svg += `<line x1="${x0 + i * s}" y1="${y0}" x2="${x0 + i * s}" y2="${y0 + b * s}" class="cell-line"/>`;
                for (let j = 1; j < b; j++) svg += `<line x1="${x0}" y1="${y0 + j * s}" x2="${x0 + a * s}" y2="${y0 + j * s}" class="cell-line"/>`;
                return svg;
            }

            // --- Слайд 1: дві кімнати ---
            window.renderRooms = function () {
                document.getElementById('rooms-scene').innerHTML =
                    cellsRect(30, 20, 4, 3, 30, 'room') + `<text x="90" y="135" class="svg-caption">Оля: 4 м × 3 м</text>`
                    + cellsRect(170, 140, 6, 2, 30, 'room b') + `<text x="260" y="228" class="svg-caption">Тарас: 6 м × 2 м</text>`;
            };

            // --- Слайд 2: заповнюємо прямокутник квадратиками 1 см (1 см = 40 px) ---
            const CELL = 40, OX = 40, OY = 20;
            let fillTimer = null;
            function fillVals() {
                return [parseInt(document.getElementById('w-a').value, 10) || 1, parseInt(document.getElementById('w-b').value, 10) || 1];
            }
            function fillSvg(a, b, n) {
                let svg = '';
                for (let x = 0; x <= 8; x++) svg += `<line x1="${OX + x * CELL}" y1="${OY}" x2="${OX + x * CELL}" y2="${OY + 5 * CELL}" class="grid-line"/>`;
                for (let y = 0; y <= 5; y++) svg += `<line x1="${OX}" y1="${OY + y * CELL}" x2="${OX + 8 * CELL}" y2="${OY + y * CELL}" class="grid-line"/>`;
                const y0 = OY + (5 - b) * CELL;
                for (let k = 0; k < n; k++) {
                    const r = Math.floor(k / a), c = k % a;
                    svg += `<rect x="${OX + c * CELL}" y="${y0 + (b - 1 - r) * CELL}" width="${CELL}" height="${CELL}" class="cell"/>`;
                }
                svg += `<rect x="${OX}" y="${y0}" width="${a * CELL}" height="${b * CELL}" class="outline"/>`;
                return svg;
            }
            function fillText(a, b, n) {
                if (n === 0) return '';
                if (n < a * b) return `Квадратиків: ${n}`;
                const rows = b === 1 ? 'ряд' : b < 5 ? 'ряди' : 'рядів';
                return `${b} ${rows} по ${a}: ${a} · ${b} = ${a * b} см²`;
            }
            function showFill(a, b, n) {
                document.getElementById('fill-scene').innerHTML = fillSvg(a, b, n);
                const el = document.getElementById('fill-count');
                el.textContent = fillText(a, b, n);
                el.classList.toggle('done', n === a * b);
            }
            window.renderFill = function () {
                clearInterval(fillTimer);
                const [a, b] = fillVals();
                document.getElementById('w-a-val').textContent = a + ' см';
                document.getElementById('w-b-val').textContent = b + ' см';
                showFill(a, b, 0);
            };
            window.fillCells = function () {
                renderFill();
                const [a, b] = fillVals();
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) { showFill(a, b, a * b); return; }
                let n = 0;
                fillTimer = setInterval(() => {
                    n++;
                    showFill(a, b, n);
                    if (n >= a * b) clearInterval(fillTimer);
                }, Math.max(60, Math.round(2400 / (a * b))));
            };

            // --- Слайд 5: периметр і площа на одному рисунку ---
            window.renderTrap = function () {
                const c = 40, x0 = 120, y0 = 20;
                document.getElementById('trap-scene').innerHTML = cellsRect(x0, y0, 4, 3, c, 'room')
                    + `<rect x="${x0}" y="${y0}" width="${4 * c}" height="${3 * c}" fill="none" stroke="var(--warning-red)" stroke-width="5"/>`
                    + `<text x="${x0 + 2 * c}" y="${y0 + 3 * c + 22}" class="len">4 см</text><text x="${x0 + 4 * c + 26}" y="${y0 + 1.5 * c + 5}" class="len">3 см</text>`
                    + `<text x="200" y="192" class="svg-caption" style="fill:var(--warning-red)">червоний край: P = 14 см</text>`
                    + `<text x="200" y="218" class="svg-caption" style="fill:var(--brand-blue)">12 клітинок усередині: S = 12 см²</text>`;
            };
