            // --- Спільне: координатний промінь від x0 з кроком px на поділку ---
            function raySvg(x0, y, px, ticks, label, endX) {
                let svg = `<line x1="${x0}" y1="${y}" x2="${endX - 6}" y2="${y}" class="ray"/>`
                        + `<path d="M${endX} ${y} L${endX - 12} ${y - 6} L${endX - 12} ${y + 6} Z" class="ray-head"/>`;
                for (let k = 0; k <= ticks; k++) {
                    const x = x0 + k * px, lab = label(k);
                    svg += `<line x1="${x}" y1="${y - (lab !== '' ? 9 : 6)}" x2="${x}" y2="${y + (lab !== '' ? 9 : 6)}" class="tick${lab !== '' ? ' big' : ''}"/>`;
                    if (lab !== '') svg += `<text x="${x}" y="${y + 27}" class="tick-num">${lab}</text>`;
                }
                return svg;
            }
            function dot(x, y, name) {
                return `<circle cx="${x}" cy="${y}" r="6" class="pt"/><text x="${x}" y="${y - 14}" class="pt-name">${name}</text>`;
            }

            // --- Слайд 1: термометр від 0 до 40, між десятками 5 проміжків ---
            window.renderThermo = function () {
                const top = 20, bottom = 210, x = 80, step = (bottom - top) / 20; // 20 проміжків по 2 °C
                let svg = `<rect x="${x - 10}" y="${top - 10}" width="20" height="${bottom - top + 20}" rx="10" class="thermo-glass"/>`
                        + `<circle cx="${x}" cy="${bottom + 22}" r="18" class="thermo-red"/>`;
                const level = bottom - 12 * step; // 24 °C — дванадцята поділка від нуля
                svg += `<rect x="${x - 5}" y="${level}" width="10" height="${bottom - level + 12}" class="thermo-red"/>`;
                for (let k = 0; k <= 20; k++) {
                    const y = bottom - k * step, big = k % 5 === 0;
                    svg += `<line x1="${x + 12}" y1="${y}" x2="${x + (big ? 32 : 24)}" y2="${y}" class="tick${big ? ' big' : ''}"/>`;
                    if (big) svg += `<text x="${x + 52}" y="${y + 5}" class="tick-num">${k * 2}</text>`;
                }
                svg += `<text x="${x + 80}" y="${level + 5}" class="pt-name" style="font-style:normal">?</text>`;
                document.getElementById('thermo-scene').innerHTML = svg;
            };

            // --- Слайд 2: стрибки коника на n одиничних відрізків ---
            const J0 = 30, JPX = 28, JY = 110;
            function jumpsScene(done) {
                const n = parseInt(document.getElementById('w-n').value, 10) || 1;
                let svg = raySvg(J0, JY, JPX, 13, (k) => k, 410)
                        + `<line x1="${J0}" y1="${JY}" x2="${J0 + JPX}" y2="${JY}" class="unit-seg"/>`
                        + `<text x="${J0 + JPX / 2}" y="${JY + 50}" class="svg-caption" style="font-size:12px;fill:var(--brand-green)">одиничний</text>`
                        + `<text x="${J0}" y="${JY - 22}" class="pt-name">O</text>`;
                for (let k = 0; k < done; k++) {
                    const a = J0 + k * JPX, b = a + JPX;
                    svg += `<path d="M${a} ${JY} Q${(a + b) / 2} ${JY - 45} ${b} ${JY}" class="jump"/>`;
                }
                if (done) svg += dot(J0 + done * JPX, JY, 'A');
                return svg;
            }
