                    <p>$\left(8^{\frac{1}{3}}\right)^3 = 8^{\frac{1}{3} \cdot 3} = 8^1 = 8$.</p>

                    <p>Так само $8^{\frac{2}{3}} = \left(8^{\frac{1}{3}}\right)^2 = \sqrt[3]{8^2} = 4$.</p>

            // --- Слайд 4: a^r для a = 4, 8, 1/4 (основа — степінь двійки, тож значення точні) ---
            const BASES = [
                { tex: '4', k: 2, d: 2 },        // 4 = 2², крок показника 1/2
                { tex: '8', k: 3, d: 3 },        // 8 = 2³, крок 1/3
                { tex: '\\frac{1}{4}', k: -2, d: 2 },
            ];
            let baseIndex = 0, powShown = false;
            const frac = (p, q) => {      // p/q у TeX, скорочено
                const g = (a, b) => (b ? g(b, a % b) : Math.abs(a));
                const c = g(p, q); p /= c; q /= c;
                if (q === 1) return `${p}`;
                return `${p < 0 ? '-' : ''}\\frac{${Math.abs(p)}}{${q}}`;
            };
            const pow2 = (e) => (e >= 0 ? `${Math.pow(2, e)}` : `\\frac{1}{${Math.pow(2, -e)}}`);
            window.setBase = function (i) {
                baseIndex = i; powShown = false;
                BASES.forEach((_, k) => document.getElementById(`b-${k}`).classList.toggle('on', k === i));
                renderPow();
            };
            window.powInput = function () { powShown = false; renderPow(); };
            window.revealPow = function () { powShown = true; renderPow(); };
            window.renderPow = function () {
                const B = BASES[baseIndex], m = parseInt(document.getElementById('p-r').value, 10);
                const r = m / B.d, e = B.k * m / B.d;      // a^r = 2^e, e — ціле
                const lab = document.getElementById('p-r-val');
                lab.innerHTML = `$${frac(m, B.d)}$`;
                const P = plane(-1.6, 1.6, -1, 9, 420, 300, 30);
                let svg = axesSvg(P, 0.5, 2, 'r', 'aʳ');
                svg += curveSvg(P, (t) => Math.pow(2, B.k * t), -1.6, 1.6);
                for (let j = -3; j <= 3; j++) {
                    const v = Math.pow(2, B.k * j / B.d);
                    if (v <= 9) svg += `<circle cx="${P.sx(j / B.d)}" cy="${P.sy(v)}" r="4" class="pt-dot"/>`;
                }
                const v = Math.pow(2, e);
                if (powShown && v <= 9) svg += `<line x1="${P.sx(r)}" y1="${P.sy(0)}" x2="${P.sx(r)}" y2="${P.sy(v)}" class="probe-line"/><circle cx="${P.sx(r)}" cy="${P.sy(v)}" r="7" class="probe"/>`;
                if (!powShown) svg += `<circle cx="${P.sx(r)}" cy="${P.sy(0)}" r="7" class="probe"/>`;
                svg += `<text x="${P.sx(B.k > 0 ? -1.5 : 1.5)}" y="${P.sy(8.2)}" class="fact-label" style="fill:var(--brand-blue);text-anchor:${B.k > 0 ? 'start' : 'end'}">${B.k > 0 ? 'a > 1: зростає' : '0 < a < 1: спадає'}</text>`;
