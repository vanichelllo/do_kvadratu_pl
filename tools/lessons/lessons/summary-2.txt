                { task: 'Кімната 6 м на 4 м, висота 3 м. Фарбують стіни; вікно й двері разом займають 4 м². На 8 м² стіни йде 1 л фарби. Скільки літрів фарби потрібно?',
                  lines: ['Периметр підлоги: 2 · (6 + 4) = 20 м — це довжина всіх стін.', 'Площа стін: 20 · 3 = 60 м² (стіни — прямокутник 20 м на 3 м).', 'Без вікна й дверей: 60 − 4 = 56 м².', 'Фарби: 56 : 8 = 7 л.', 'Відповідь: 7 л.'] },
                { task: 'Коробку 30 см на 20 см на 10 см обклеюють стрічкою по всіх ребрах. Скільки стрічки потрібно і скільки літрів уміщує коробка?',
                  lines: ['Ребер кожного виміру — по 4: 4 · (30 + 20 + 10) = 240 см стрічки.', '240 см = 2 м 40 см.', 'Об\'єм: 30 · 20 · 10 = 6 000 см³.', '1 л = 1 000 см³, тож 6 000 см³ = 6 л.', 'Відповідь: 2 м 40 см стрічки; 6 л.'] },
            ];
            let taskIndex = 0, shownSol = 0;
            window.renderSolution = function () {
                const T = TASKS[taskIndex];
                document.getElementById('sol-task').textContent = T.task;
                const box = document.getElementById('sol-lines');
                box.innerHTML = '';
                T.lines.slice(0, shownSol).forEach(l => { const d = document.createElement('div'); d.className = 'sol-line'; d.textContent = l; box.appendChild(d); });
            };
            window.solStep = function () { shownSol = Math.min(shownSol + 1, TASKS[taskIndex].lines.length); renderSolution(); };
            window.solReset = function () { shownSol = 0; renderSolution(); };
            window.solNext = function () { taskIndex = (taskIndex + 1) % TASKS.length; shownSol = 0; renderSolution(); };
