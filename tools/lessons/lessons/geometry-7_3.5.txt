            function vName(p, c, name) {
                const dx = p[0] - c[0], dy = p[1] - c[1], L = Math.hypot(dx, dy) || 1;
                return `<text x="${f1(p[0] + dx / L * 18)}" y="${f1(p[1] + dy / L * 18 + 6)}" class="v-name">${name}</text>`;
            }
            const names = (P, N) => { const c = [(P[0][0] + P[1][0] + P[2][0]) / 3, (P[0][1] + P[1][1] + P[2][1]) / 3]; return P.map((p, k) => vName(p, c, N[k])).join(''); };
            // Сектор кута при вершині V між напрямками на P і Q (менший з кутів)
            function angleArc(V, P, Q, r, cls) {
                const a1 = Math.atan2(P[1] - V[1], P[0] - V[0]), a2 = Math.atan2(Q[1] - V[1], Q[0] - V[0]);
                let d = a2 - a1;
                while (d > Math.PI) d -= 2 * Math.PI;
                while (d < -Math.PI) d += 2 * Math.PI;
                const s = [V[0] + r * Math.cos(a1), V[1] + r * Math.sin(a1)], e = [V[0] + r * Math.cos(a1 + d), V[1] + r * Math.sin(a1 + d)];
                return `<path d="M${f1(V[0])} ${f1(V[1])} L${f1(s[0])} ${f1(s[1])} A${r} ${r} 0 0 ${d > 0 ? 1 : 0} ${f1(e[0])} ${f1(e[1])} Z" class="angle-arc ${cls}"/>`;
            }
            // Кут при V між P і Q у градусах
            const ang = (V, P, Q) => {
                const a = [P[0] - V[0], P[1] - V[1]], b = [Q[0] - V[0], Q[1] - V[1]];
                return Math.acos((a[0] * b[0] + a[1] * b[1]) / Math.hypot(...a) / Math.hypot(...b)) * 180 / Math.PI;
            };
            // Точка на бісектрисі кута (для підпису)
            const inside = (V, P, Q, r) => {
                const u = [P[0] - V[0], P[1] - V[1]], w = [Q[0] - V[0], Q[1] - V[1]], lu = Math.hypot(...u), lw = Math.hypot(...w);
                const d = [u[0] / lu + w[0] / lw, u[1] / lu + w[1] / lw], ld = Math.hypot(...d) || 1;
                return [V[0] + d[0] / ld * r, V[1] + d[1] / ld * r + 5];
            };
