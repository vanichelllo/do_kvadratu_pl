            function buildVals() { return [parseInt(document.getElementById('w-c').value, 10), parseInt(document.getElementById('w-al').value, 10), parseFloat(document.getElementById('w-s').value)]; }
            // Вершини C: для «кута між сторонами» — одна; для «не між» — точки перетину променя з колом радіуса BC навколо B
            function apexes(c, al, s) {
                const u = [Math.cos(rad(al)), Math.sin(rad(al))];
                if (mode === 0) return [[s * u[0], s * u[1]]];
                const p = c * Math.cos(rad(al)), D = s * s - c * c * Math.sin(rad(al)) ** 2;
                if (D < 0) return [];
                const ts = D === 0 ? [p] : [p - Math.sqrt(D), p + Math.sqrt(D)];
                return ts.filter(t => t > 1e-6).map(t => [t * u[0], t * u[1]]);
            }
            function buildScene(show) {
                const [c, al, s] = buildVals(), S = 36, x0 = 40, y0 = 230, P = (q) => [x0 + q[0] * S, y0 - q[1] * S];
                const A = P([0, 0]), B = P([c, 0]), far = P([11 * Math.cos(rad(al)), 11 * Math.sin(rad(al))]);
                let svg = seg(A, far, 'ray') + angleArc(A, B, far, 30, 'g');
                if (mode === 1) svg += `<circle cx="${f1(B[0])}" cy="${f1(B[1])}" r="${f1(s * S)}" class="arc-guide"/>`;
                const C = apexes(c, al, s);
                if (show) C.forEach((q, k) => { svg += poly([A, B, P(q)], k ? 'tri r' : 'tri g') + dot(P(q)) + `<text x="${f1(P(q)[0] - 14)}" y="${f1(P(q)[1] - 8)}" class="v-name">C${C.length > 1 ? (k ? '₂' : '₁') : ''}</text>`; });
                svg += seg(A, B, 'hl') + dot(A) + dot(B) + `<text x="${f1(A[0])}" y="${y0 + 24}" class="v-name">A</text><text x="${f1(B[0])}" y="${y0 + 24}" class="v-name">B</text>`;
                if (mode === 0 && !show) svg += seg(A, P(C[0]), 'hl');
                return svg;
            }
