            const T1 = { h: ['Зошитів', 'Вартість, грн'], cols: [[1, 15], [2, 30], [4, 60], [6, 90]] };
            const T2 = { h: ['Швидкість, км/год', 'Час, год'], cols: [[20, 6], [40, 3], [60, 2], [120, 1]] };
            const ORDER = [[0, 1], [0, 2], [0, 3], [1, 1], [1, 2], [1, 3]];
            let shownCells = 0;
            const tbl = (T, ti) => `<table class="tbl"><tr><th>${T.h[0]}</th>${T.cols.map(c => `<td>${c[0]}</td>`).join('')}</tr><tr><th>${T.h[1]}</th>${T.cols.map((c, k) => {
                const idx = ORDER.findIndex(([t, j]) => t === ti && j === k);
                return `<td class="${idx >= 0 ? 'new' : ''}">${idx < 0 || idx < shownCells ? c[1] : '?'}</td>`;
            }).join('')}</tr></table>`;
            window.renderCells = function () {
                document.getElementById('tbl-1').innerHTML = tbl(T1, 0);
                document.getElementById('tbl-2').innerHTML = tbl(T2, 1);
                const n = document.getElementById('tbl-note');
                n.innerHTML = shownCells >= 3 ? 'Зошити: вартість : кількість $= 15$ — частка стала' : '';
                if (shownCells >= 6) n.innerHTML += '<br>Дорога: швидкість $\\cdot$ час $= 120$ — добуток сталий';
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([n]);
            };
            window.revealCell = function () { shownCells = Math.min(shownCells + 1, 6); renderCells(); };
            window.resetCells = function () { shownCells = 0; renderCells(); };
