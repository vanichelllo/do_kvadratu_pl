            // --- Слайд 2: s = v · t ---
            const val = (id) => parseInt(document.getElementById(id).value, 10) || 1;
            window.renderTrip = function () {
                const v = val('w-v'), t = val('w-t'), s = v * t;
                document.getElementById('w-v-val').textContent = `${v} км/год`;
                document.getElementById('w-t-val').textContent = `${t} год`;
                document.getElementById('trip-answer').classList.remove('shown');
                // Шкала до 500 км; машина стоїть у кінці шляху, але число шляху ховаємо до кліку
                const x0 = 20, scale = 400 / 500, xEnd = x0 + s * scale;
                let svg = `<line x1="${x0}" y1="110" x2="${x0 + 400}" y2="110" class="road"/>`;
                for (let km = 0; km <= 500; km += 50) {
                    const x = x0 + km * scale;
                    svg += `<line x1="${x}" y1="104" x2="${x}" y2="116" class="road-tick"/><text x="${x}" y="134" class="road-label">${km}</text>`;
                }
                svg += `<text x="${x0 + 400}" y="152" class="road-label">км</text>`;
                for (let h = 1; h <= t; h++) {
                    const xa = x0 + (h - 1) * v * scale, xb = x0 + h * v * scale;
                    svg += `<line x1="${xa + 2}" y1="92" x2="${xb - 2}" y2="92" class="trip"/><text x="${(xa + xb) / 2}" y="82" class="trip-cap">${h} год</text>`;
                }
                svg += `<rect x="${xEnd - 22}" y="40" width="44" height="20" rx="6" class="car-body"/><rect x="${xEnd - 12}" y="30" width="24" height="14" rx="4" class="car-body"/>`
                     + `<circle cx="${xEnd - 12}" cy="62" r="6" class="car-wheel"/><circle cx="${xEnd + 12}" cy="62" r="6" class="car-wheel"/>`;
                document.getElementById('trip-scene').innerHTML = svg;
            };
            window.revealTrip = function () {
                const v = val('w-v'), t = val('w-t');
                document.getElementById('trip-s').textContent = `s = ${v} · ${t} = ${v * t} км`;
                document.getElementById('trip-answer').classList.add('shown');
            };

            // --- Слайд 4: покрокове обчислення виразів ---
            // Кожен приклад — рядки обчислення; [[…]] — дія, яку виконуємо на цьому кроці; note — пояснення
            const EXAMPLES = [
                { lines: [
                    ['120 : [[(17 − 5)]] · 3 + 4', 'Спершу дужки'],
                    ['[[120 : 12]] · 3 + 4', 'Ділення й множення — зліва направо'],
                    ['[[10 · 3]] + 4', 'Тепер множення'],
                    ['[[30 + 4]]', 'Наостанок додавання'],
                    ['34', 'Відповідь: 34'],
                ] },
                { lines: [
                    ['5 + [[45 : 9]] · 2 − 3', 'Дужок немає — спершу ділення (воно лівіше)'],
                    ['5 + [[5 · 2]] − 3', 'Потім множення'],
                    ['[[5 + 10]] − 3', 'Додавання й віднімання — зліва направо'],
                    ['[[15 − 3]]', ''],
                    ['12', 'Відповідь: 12'],
                ] },
                { lines: [
                    ['4a + 25, якщо a = 15', 'Вираз з буквою: спершу підставляємо число'],
                    ['[[4 · 15]] + 25', '4a — це 4 · a'],
                    ['[[60 + 25]]', ''],
                    ['85', 'Відповідь: при a = 15 значення виразу 85'],
                ] },
            ];
            let exIndex = 0, shownLines = 1;
            const esc = (s) => s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
            window.renderEval = function () {
                const ex = EXAMPLES[exIndex], box = document.getElementById('eval-lines');
                box.innerHTML = ex.lines.slice(0, shownLines).map(([line], k) => {
                    const html = esc(line).replace(/\[\[(.+?)\]\]/g, k === shownLines - 1 ? '<mark>$1</mark>' : '$1');
                    return `<div class="eval-line${k === ex.lines.length - 1 ? ' done' : ''}">${k ? '= ' : ''}${html}</div>`;
                }).join('');
                document.getElementById('eval-note').textContent = ex.lines[shownLines - 1][1];
            };
            window.evalStep = function () { shownLines = Math.min(shownLines + 1, EXAMPLES[exIndex].lines.length); renderEval(); };
            window.evalReset = function () { shownLines = 1; renderEval(); };
            window.evalNext = function () { exIndex = (exIndex + 1) % EXAMPLES.length; shownLines = 1; renderEval(); };
