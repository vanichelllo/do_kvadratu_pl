            // --- Слайд 2: обхід многокутника (координати в см, 1 см = 40 px) ---
            const SC = 40, BX = 90, BY = 230;
            const SHAPES = [
                { name: 'Трикутник', pts: [[0, 0], [4, 0], [0, 3]] },
                { name: 'Прямокутник', pts: [[0, 0], [6, 0], [6, 3], [0, 3]] },
                { name: 'Квадрат', pts: [[0, 0], [4, 0], [4, 4], [0, 4]] },
                { name: 'Шестикутник', pts: [[0, 0], [5, 0], [5, 2], [2, 2], [2, 4], [0, 4]] },
            ];
            let shapeIdx = 0, walkAnim = null;
            const px = (p) => ({ x: BX + p[0] * SC, y: BY - p[1] * SC });
            function sideLens(pts) {
                return pts.map((p, i) => { const q = pts[(i + 1) % pts.length]; return Math.round(Math.hypot(q[0] - p[0], q[1] - p[1])); });
            }
            function shapeSvg(done, runner) {
                const pts = SHAPES[shapeIdx].pts, P = pts.map(px), lens = sideLens(pts);
                let svg = `<polygon points="${P.map(p => p.x + ',' + p.y).join(' ')}" class="poly"/>`;
                P.forEach((a, i) => {
                    const b = P[(i + 1) % P.length];
                    if (i < done) svg += `<line x1="${a.x}" y1="${a.y}" x2="${b.x}" y2="${b.y}" class="side-done"/>`;
                    // Підпис — назовні від середини сторони (вершини задано проти годинникової стрілки)
                    const mx = (a.x + b.x) / 2, my = (a.y + b.y) / 2;
                    const c = pts[i], d = pts[(i + 1) % pts.length], dl = Math.hypot(d[0] - c[0], d[1] - c[1]) || 1;
                    const nx = (d[1] - c[1]) / dl, ny = (d[0] - c[0]) / dl;   // нормаль (dy, −dx), вісь y на екрані — донизу
                    svg += `<text x="${(mx + nx * 20).toFixed(1)}" y="${(my + ny * 20 + 5).toFixed(1)}" class="len">${lens[i]} см</text>`;
                });
                if (runner) svg += `<circle cx="${runner.x.toFixed(1)}" cy="${runner.y.toFixed(1)}" r="9" class="runner"/>`;
                return svg;
            }
            function sumText(done) {
                const lens = sideLens(SHAPES[shapeIdx].pts);
                if (done === 0) return '';
                const part = lens.slice(0, done), s = part.reduce((a, b) => a + b, 0);
                return part.join(' + ') + ' = ' + s + ' см' + (done === lens.length ? ' — це периметр!' : '');
            }
            function showSum(done) {
                const el = document.getElementById('walk-sum'), n = SHAPES[shapeIdx].pts.length;
                el.textContent = sumText(done);
                el.classList.toggle('done', done === n);
            }
            window.renderShape = function () {
                cancelAnimationFrame(walkAnim);
                document.getElementById('walk-scene').innerHTML = shapeSvg(0, px(SHAPES[shapeIdx].pts[0]));
                showSum(0);
            };
            window.setShape = function (i) {
                shapeIdx = i;
                SHAPES.forEach((_, j) => document.getElementById('shape-' + j).classList.toggle('on', j === i));
                renderShape();
            };
            window.walkShape = function () {
                renderShape();
                const pts = SHAPES[shapeIdx].pts, P = pts.map(px), n = P.length;
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) { document.getElementById('walk-scene').innerHTML = shapeSvg(n, P[0]); showSum(n); return; }
                const t0 = performance.now(), dur = 900 * n;
                let shown = -1;
                const step = (now) => {
                    const k = Math.min(1, Math.max(0, (now - t0) / dur)), pos = k * n, i = Math.min(n - 1, Math.floor(pos)), f = pos - i;
                    const done = k >= 1 ? n : i, a = P[i], b = P[(i + 1) % n];
                    const r = k >= 1 ? P[0] : { x: a.x + (b.x - a.x) * f, y: a.y + (b.y - a.y) * f };
                    document.getElementById('walk-scene').innerHTML = shapeSvg(done, r);
                    if (done !== shown) { shown = done; showSum(done); }
                    if (k < 1) walkAnim = requestAnimationFrame(step);
                };
                walkAnim = requestAnimationFrame(step);
            };

            // --- Слайд 4: периметр прямокутника ---
            const STEPS = [
                ['Ширина: $8 - 3 = $ [[$5$ см]].', '«На $3$ см менша» — віднімаємо'],
                ['Довжина й ширина разом: $8 + 5 = $ [[$13$ см]].', 'Це половина периметра'],
                ['$P = 2 \\cdot 13 = $ [[$26$ см]].', 'Дві довжини й дві ширини'],
                ['Відповідь: $P = 26$ см.', 'Перевірка: $8 + 5 + 8 + 5 = 26$ ✓'],
            ];
