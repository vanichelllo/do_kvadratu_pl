            // Плитка з n часточок; перші m зафарбовано класом cls
            function barSvg(x, y, w, h, n, m, cls) {
                let svg = '';
                for (let k = 0; k < n; k++) {
                    svg += `<rect x="${(x + k * w / n).toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="strip-cell"/>`;
                    if (k < m) svg += `<rect x="${(x + k * w / n).toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="${cls}" style="stroke:var(--text-dark);stroke-width:2"/>`;
                }
                return svg;
            }
            // Мішане число в SVG: ціла частина й маленький дріб праворуч
            function mixedSvg(x, y, q, r, n) {
                let svg = q > 0 ? `<text x="${x}" y="${y + 10}" class="mix-whole">${q}</text>` : '';
                if (r > 0) svg += `<text x="${x + 14}" y="${y - 4}" class="mix-num">${r}</text><line x1="${x + 4}" y1="${y + 1}" x2="${x + 24}" y2="${y + 1}" class="mix-bar"/><text x="${x + 14}" y="${y + 20}" class="mix-den">${n}</text>`;
                return svg;
            }
            // Плитки для m часточок по n у кожній: повні — сині, неповна — зелена
            function barsSvg(n, m, x0, y0, colW, rowH, perRow, w, h, captions) {
                const bars = Math.max(1, Math.ceil(m / n));
                let svg = '';
                for (let i = 0; i < bars; i++) {
                    const filled = Math.min(n, m - i * n), x = x0 + (i % perRow) * colW, y = y0 + Math.floor(i / perRow) * rowH;
                    svg += barSvg(x, y, w, h, n, filled, filled === n ? 'strip-a' : 'strip-b');
                    if (captions) svg += filled === n
                        ? `<text x="${x + w / 2}" y="${y + h + 20}" class="svg-caption" style="fill:var(--brand-blue)">1 ціла</text>`
                        : `<text x="${x + w / 2}" y="${y + h + 20}" class="svg-caption" style="fill:var(--brand-green)">${filled}/${n}</text>`;
                }
                return svg;
            }
            // «1 ціла», «2 цілі», «5 цілих»
            function wholeWord(k) {
                const d = k % 10, h = k % 100;
                return (d === 1 && h !== 11) ? 'ціла' : (d >= 2 && d <= 4 && (h < 12 || h > 14)) ? 'цілі' : 'цілих';
            }
