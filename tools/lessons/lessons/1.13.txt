            // --- Слайд 2: стрічки Тараса, Олі й «разом» проти мети 50 ---
            const UNIT = 5, X0 = 80, TOTAL = 50, MORE = 8;
            function barsSvg(x, ok) {
                const bx = (y, from, units, cls, label, dark) => `<rect x="${X0 + from * UNIT}" y="${y}" width="${units * UNIT}" height="34" class="${cls}"/>`
                    + (units * UNIT >= 22 ? `<text x="${X0 + (from + units / 2) * UNIT}" y="${y + 23}" class="bar-in${dark ? ' dark' : ''}">${label}</text>` : '');
                let svg = `<g class="${ok ? 'bar-ok' : ''}">`;
                svg += `<text x="8" y="57" class="bar-label">Тарас</text>` + bx(34, 0, x, 'bar-x', 'x');
                svg += `<text x="8" y="117" class="bar-label">Оля</text>` + bx(94, 0, x, 'bar-x', 'x') + bx(94, x, MORE, 'bar-plus', '8', true);
                svg += `<text x="8" y="177" class="bar-label">Разом</text>` + bx(154, 0, x, 'bar-x', 'x') + bx(154, x, x, 'bar-x', 'x')
                     + bx(154, 2 * x, MORE, 'bar-plus', '8', true);
                svg += '</g>';
                const tx = X0 + TOTAL * UNIT;
                svg += `<line x1="${tx}" y1="140" x2="${tx}" y2="205" class="target"/><text x="${tx}" y="228" class="target-label">50</text>`;
                return svg;
            }
            function barsX() { return parseInt(document.getElementById('w-x').value, 10) || 5; }
            window.renderBars = function (ok) {
                const x = barsX();
                document.getElementById('w-x-val').textContent = x;
                document.getElementById('bars-scene').innerHTML = barsSvg(x, ok === true);
                if (ok !== true) document.getElementById('bars-readout').textContent = '';
            };
            window.checkBars = function () {
                const x = barsX(), sum = 2 * x + MORE, out = document.getElementById('bars-readout');
                renderBars(sum === TOTAL);
                out.className = 'readout ' + (sum === TOTAL ? 'ok' : 'bad');
                out.textContent = `${x} + ${x + MORE} = ${sum}` + (sum === TOTAL ? ' ✓ Знайшли!' : sum < TOTAL ? ' — замало' : ' — забагато');
            };
            // Анімація: x зростає від 5, доки разом не стане 50 (вчитель запускає кнопкою)
            let barsTimer = null;
            window.runBars = function () {
                clearInterval(barsTimer);
                const slider = document.getElementById('w-x');
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) { slider.value = (TOTAL - MORE) / 2; checkBars(); return; }
                slider.value = 5; renderBars();
                barsTimer = setInterval(() => {
                    const x = barsX();
                    if (2 * x + MORE >= TOTAL) { clearInterval(barsTimer); checkBars(); return; }
                    slider.value = x + 1; renderBars();
                    document.getElementById('bars-readout').textContent = `${x + 1} + ${x + 1 + MORE} = ${2 * (x + 1) + MORE}`;
                    document.getElementById('bars-readout').className = 'readout';
                }, 160);
            };
