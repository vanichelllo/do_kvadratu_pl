                    <p>1) $a^m \cdot a^n = a^{m + n}$ — показники <b>додаємо</b>.</p>
                    <p>2) $a^m : a^n = a^{m - n}$ ($a \ne 0$, $m \gt n$) — показники <b>віднімаємо</b>.</p>
                    <p>3) $(a^m)^n = a^{mn}$ — показники <b>множимо</b>.</p>
                    <p>4) $(ab)^n = a^n b^n$, $\;\left(\dfrac{a}{b}\right)^n = \dfrac{a^n}{b^n}$ ($b \ne 0$).</p>
                </div>
                <div class="box example">
                    <div class="box-title">Чому $a^m \cdot a^n = a^{m + n}$</div>
                    <p>$a^3 \cdot a^2 = (a \cdot a \cdot a) \cdot (a \cdot a) = a^5$: усього $3 + 2$ множники.</p>
                    <p>$(a^3)^2 = a^3 \cdot a^3 = a^6$: двічі по $3$ множники.</p>
                    <p>Окремо домовились: $a^0 = 1$ при $a \ne 0$ (тоді $a^n : a^n = a^0 = 1$).</p>
                </div>
            </div>

            const typeset = (el) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([el]); };

            // --- Слайд 2: клік по «?» відкриває значення ---

            let op = 0, powShown = false;
            window.setOp = function (k) {
                op = k;
                [0, 1, 2].forEach(j => document.getElementById('op-' + j).classList.toggle('on', j === k));
                renderPow();
            };
            function powVals() {
                const M = document.getElementById('w-m'), N = document.getElementById('w-n');
                if (op === 1) {
                    M.min = 2;
                    if (parseInt(M.value, 10) < 2) M.value = 2;
                    N.max = parseInt(M.value, 10) - 1;
                    if (parseInt(N.value, 10) > parseInt(N.max, 10)) N.value = N.max;
                } else { M.min = 1; N.max = 4; }
                return [parseInt(M.value, 10), parseInt(N.value, 10)];
            }
            const chip = (cls, crossed) => `<span class="pow-chip ${cls}"${crossed ? ' style="opacity:0.35;text-decoration:line-through"' : ''}>$a$</span>`;
            window.renderPow = function () {
                const [m, n] = powVals();
                document.getElementById('w-m-val').textContent = m;
                document.getElementById('w-n-val').textContent = n;
                powShown = false;
                const task = document.getElementById('pow-task');
                task.innerHTML = op === 0 ? `$a^{${m}} \\cdot a^{${n}} = a^{\\,?}$` : op === 1 ? `$a^{${m}} : a^{${n}} = a^{\\,?}$` : `$(a^{${m}})^{${n}} = a^{\\,?}$`;
                document.getElementById('pow-chips').innerHTML = '';
                document.getElementById('pow-result').innerHTML = '';
                document.getElementById('pow-readout').innerHTML = 'Який показник вийде?';
                typeset(task);
            };
            window.revealPow = function () {
                const [m, n] = powVals(), box = document.getElementById('pow-chips'), res = document.getElementById('pow-result');
                let html = '', r;
                if (op === 0) {
                    html = `<div class="pow-row">${Array(m).fill(chip('', false)).join('')}<b style="align-self:center">·</b>${Array(n).fill(chip('b', false)).join('')}</div>`;
                    r = m + n; res.innerHTML = `$${m} + ${n} = ${r}$ множників: $a^{${m}} \\cdot a^{${n}} = a^{${r}}$`;
                } else if (op === 1) {
                    html = `<div class="pow-row">${Array(m).fill(0).map((_, k) => chip('', k >= m - n)).join('')}</div><div class="pow-row">${Array(n).fill(chip('b', true)).join('')}</div>`;
