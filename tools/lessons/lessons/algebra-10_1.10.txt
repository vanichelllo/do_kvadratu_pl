            // Числова пряма: points — [{x, filled}], signs — знаки на інтервалах зліва направо, shade — індекси інтервалів
            function lineSvg(pts, signs, shade = [], shadePts = []) {
                const W = 420, y = 90, x0 = 30, x1 = 390, n = pts.length, pos = (k) => x0 + (x1 - x0) * (k + 1) / (n + 1);
                let s = '';
                const edges = [x0 - 10, ...pts.map((_, k) => pos(k)), x1 + 10];
                shade.forEach(i => s += `<rect x="${edges[i]}" y="${y - 12}" width="${edges[i + 1] - edges[i]}" height="24" class="shade"/>`);
                s += `<line x1="${x0 - 10}" y1="${y}" x2="${x1 + 10}" y2="${y}" class="axis-l"/><path d="M${x1 + 14} ${y} l-10 -5 v10 z" fill="currentColor"/>`;
                signs.forEach((sg, i) => { if (sg) s += `<text x="${(edges[i] + edges[i + 1]) / 2}" y="${y - 22}" class="sign ${sg === '+' ? 'p' : 'm'}">${sg === '+' ? '+' : '−'}</text>`; });
                pts.forEach((p, k) => {
                    s += `<circle cx="${pos(k)}" cy="${y}" r="6" class="${p.filled || shadePts.includes(k) ? 'pt-in' : 'pt-out'}"/><text x="${pos(k)}" y="${y + 28}" class="lbl">${p.t}</text>`;
                });
                return s;
            }
