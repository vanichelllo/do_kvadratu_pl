            // Підпис вершини: трохи назовні від центру фігури
            function vName(p, c, name) {
                const dx = p[0] - c[0], dy = p[1] - c[1], L = Math.hypot(dx, dy) || 1;
                return `<text x="${f1(p[0] + dx / L * 18)}" y="${f1(p[1] + dy / L * 18 + 6)}" class="v-name">${name}</text>`;
            }
            // Паралелограм ABCD: A — початок, AB = a по горизонталі, AD = b під кутом ang; s — масштаб
            function parPts(a, b, ang, s, x0, y0) {
                const A = [x0, y0], B = [x0 + a * s, y0], D = [x0 + b * s * Math.cos(rad(ang)), y0 - b * s * Math.sin(rad(ang))];
                return [A, B, [B[0] + D[0] - A[0], B[1] + D[1] - A[1]], D];
            }
            function parSvg(P, opts) {
                const [A, B, C, D] = P, c = [(A[0] + C[0]) / 2, (A[1] + C[1]) / 2];
                let s = `<polygon points="${P.map(p => p.map(f1).join(',')).join(' ')}" class="side"/>`;
                if (opts.diag) s += `<line x1="${f1(A[0])}" y1="${f1(A[1])}" x2="${f1(C[0])}" y2="${f1(C[1])}" class="diag"/><line x1="${f1(B[0])}" y1="${f1(B[1])}" x2="${f1(D[0])}" y2="${f1(D[1])}" class="diag2"/>`;

                return s + ['A', 'B', 'C', 'D'].map((n, k) => vName(P[k], c, n)).join('');
            }

            function parVals() { return [parseInt(document.getElementById('w-ang').value, 10), parseInt(document.getElementById('w-side').value, 10)]; }
            function parScene(ang, b, show) {
                const a = 4, s = Math.min(330 / (a + b * Math.abs(Math.cos(rad(ang)))), 200 / (b * Math.sin(rad(ang)) || 1), 52);
                const minX = Math.min(0, b * Math.cos(rad(ang))) * s;
                const width = (a + b * Math.abs(Math.cos(rad(ang)))) * s, height = b * Math.sin(rad(ang)) * s;
                const x0 = 210 - width / 2 - minX, y0 = 135 + height / 2;
                const P = parPts(a, b, ang, s, x0, y0);
                let svg = parSvg(P, { diag: true, right: ang === 90 });
                if (show && b === a) {
                    // однакові позначки на сторонах ромба
                    for (let k = 0; k < 4; k++) {
                        const p = P[k], q = P[(k + 1) % 4], m = [(p[0] + q[0]) / 2, (p[1] + q[1]) / 2];
                        const nx = -(q[1] - p[1]), ny = q[0] - p[0], L = Math.hypot(nx, ny);
                        svg += `<line x1="${f1(m[0] - nx / L * 7)}" y1="${f1(m[1] - ny / L * 7)}" x2="${f1(m[0] + nx / L * 7)}" y2="${f1(m[1] + ny / L * 7)}" class="tick-mark"/>`;
                    }
                }
                return svg;
            }
            function parInfo(ang, b) {
                const a = 4, c = Math.cos(rad(ang)), sn = Math.sin(rad(ang));
                const d1 = Math.hypot(a + b * c, b * sn), d2 = Math.hypot(b * c - a, b * sn);
                const dot = (a + b * c) * (b * c - a) + b * sn * b * sn;
                const between = Math.acos(Math.min(1, Math.abs(dot) / (d1 * d2))) * 180 / Math.PI;
                let name = 'паралелограм';
                if (ang === 90 && b === a) name = 'квадрат'; else if (ang === 90) name = 'прямокутник'; else if (b === a) name = 'ромб';
                return { name, d1, d2, between };
            }
            window.renderPar = function () {
                const [ang, b] = parVals();
                document.getElementById('w-ang-val').textContent = ang + '°';
                document.getElementById('w-side-val').textContent = b;
                parShown = false;
                document.getElementById('par-readout').innerHTML = 'Сторона $AB = 4$. Яка це фігура?';
                typeset(document.getElementById('par-readout'));
                document.getElementById('par-scene').innerHTML = parScene(ang, b, false);
            };
            window.revealPar = function () {
                const [ang, b] = parVals(), I = parInfo(ang, b), out = document.getElementById('par-readout');
                const eq = Math.abs(I.d1 - I.d2) < 1e-9, perp = Math.abs(I.between - 90) < 1e-6;
                document.getElementById('par-scene').innerHTML = parScene(ang, b, true);
                out.innerHTML = `<span class="name">Це ${I.name}.</span><br>Діагоналі: $${I.d1.toFixed(2).replace('.', '{,}')}$ і $${I.d2.toFixed(2).replace('.', '{,}')}$ — ${eq ? '<b>рівні</b>' : 'не рівні'}; кут між ними — $${Math.round(I.between)}^\\circ$${perp ? ' — <b>перпендикулярні</b>' : ''}.`;
                typeset(out);
            };

            // --- Слайд 5: прямокутник з кутом AOB = 60° ---
