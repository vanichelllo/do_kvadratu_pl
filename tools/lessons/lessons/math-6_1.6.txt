            // --- Слайди 2 і 5: дві жабки стрибають по двох числових доріжках ---
            const gcd = (x, y) => (y ? gcd(y, x % y) : x);
            const lcm = (x, y) => x / gcd(x, y) * y;
            const FX0 = 40, FW = 422, YA = 84, YB = 156;
            // Довжина доріжок: кілька спільних кратних, але так, щоб підписи не злипались
            function frogLength(a, b, big) {
                const l = lcm(a, b), cap = Math.min(60, Math.floor(FW * Math.min(a, b) / (big ? 30 : 20)));
                return l * Math.max(1, Math.min(Math.floor(cap / l), Math.round(48 / l)));
            }
            function frogIcon(x, y, cls) {
                return `<g transform="translate(${x} ${y})"><ellipse cx="0" cy="0" rx="13" ry="9" class="${cls}"/>`
                     + `<circle cx="-6.5" cy="-8" r="5" class="${cls}"/><circle cx="6.5" cy="-8" r="5" class="${cls}"/>`
                     + `<circle cx="-6.5" cy="-8" r="2.8" class="frog-eye"/><circle cx="6.5" cy="-8" r="2.8" class="frog-eye"/>`
                     + `<circle cx="-6" cy="-8" r="1.4" class="frog-pupil"/><circle cx="7" cy="-8" r="1.4" class="frog-pupil"/>`
                     + `<path d="M-5 2 Q0 6 5 2" class="frog-mouth"/></g>`;
            }
            // o.A, o.B — показати стрибки жабок; o.common — підсвітити спільні точки; o.marks — [[число, клас]];
            // o.big — великий шрифт (вузький екран); o.animA, o.animB, o.animC — анімувати щойно відкрите
            function frogsSvg(a, b, L, o) {
                const u = FW / L, X = (m) => FX0 + m * u, fs = o.big ? 21 : 15, r = o.big ? 5.5 : 4.5, l = lcm(a, b);
                const after = 0.12 * Math.max(o.animA ? Math.floor(L / a) : 0, o.animB ? Math.floor(L / b) : 0);
                let s = '';
                if (o.common) for (let c = l; c <= L; c += l) {
                    const w = Math.min(30, 0.9 * l * u);
                    s += `<rect x="${X(c) - w / 2}" y="${YA - 16}" width="${w}" height="${YB - YA + 32}" rx="9" class="band${c === l ? ' first' : ''}${o.animC ? ' pop' : ''}"${o.animC ? ` style="animation-delay:${after}s"` : ''}/>`;
                }
                (o.marks || []).forEach(([m, cls]) => {
                    s += `<rect x="${X(m) - 15}" y="${YA - 16}" width="30" height="${YB - YA + 32}" rx="9" class="band ${cls}"/>`;
                });
                [YA, YB].forEach((y) => {
                    s += `<line x1="${FX0}" y1="${y}" x2="${FX0 + FW + 8}" y2="${y}" class="track"/>`
                       + `<path d="M${FX0 + FW + 16} ${y} l-10 -5 v10 z" class="track-arrow"/>`;
                    if (u >= 6) for (let m = 1; m <= L; m++) s += `<line x1="${X(m)}" y1="${y - 4}" x2="${X(m)}" y2="${y + 4}" class="unit"/>`;
                });
                // Одна жабка: дуги вгору (dir = −1) чи вниз (dir = 1); точки приземлення й підписи — між доріжками
                const jumps = (step, y, dir, k, show, anim) => {
                    const ly = dir < 0 ? y + fs + 6 : y - 9;
                    let t = `<circle cx="${X(0)}" cy="${y}" r="${r}" class="dot-0"/><text x="${X(0)}" y="${ly}" class="lab z" style="font-size:${fs}px">0</text>`;
                    if (!show) return t;
                    for (let m = step, i = 0; m <= L; m += step, i++) {
                        const w = step * u, h = Math.min(40, 0.3 * w + 8), d = anim ? i * 0.12 : 0;
                        const isC = o.common && m % l === 0;
                        t += `<path d="M${X(m - step)} ${y} Q${X(m - step / 2)} ${y + dir * 2 * h} ${X(m)} ${y}" class="hop ${k}${anim ? ' anim' : ''}"${anim ? ` style="animation-delay:${d}s"` : ''}/>`
                           + `<circle cx="${X(m)}" cy="${y}" r="${r}" class="dot-${k}${anim ? ' pop' : ''}"${anim ? ` style="animation-delay:${d + 0.25}s"` : ''}/>`
                           + `<text x="${X(m)}" y="${ly}" class="lab ${k}${isC ? ' common' : ''}${anim ? ' pop' : ''}" style="font-size:${fs}px${anim ? `;animation-delay:${d + 0.25}s` : ''}">${m}</text>`;
                    }
                    return t;
                };
                s += jumps(a, YA, -1, 'a', o.A, o.animA) + jumps(b, YB, 1, 'b', o.B, o.animB);
                // Жабки й довжина стрибка — ліворуч від доріжок
                s += frogIcon(18, YA - 9, 'frog-a') + `<text x="18" y="${YA - 28}" class="lab a" style="font-size:${fs}px">+${a}</text>`
                   + frogIcon(18, YB - 9, 'frog-b') + `<text x="18" y="${YB + fs + 8}" class="lab b" style="font-size:${fs}px">+${b}</text>`;
                if (o.common) {
                    const x = X(l), anchor = x > 400 ? 'end' : x < 90 ? 'start' : 'middle';
                    const tx = anchor === 'end' ? x + 14 : anchor === 'start' ? x - 14 : x;
                    s += `<text x="${tx}" y="${YB + 62}" class="mark-text${o.animC ? ' pop' : ''}" style="font-size:${fs}px;text-anchor:${anchor}${o.animC ? `;animation-delay:${after}s` : ''}">найменше ★</text>`;
                }
                return s;
            }

            // --- Слайд 2: учитель відкриває стрибки й спільні точки; повзунок усе ховає знову ---
            let frog = { A: false, B: false, C: false };
            function drawFrogs(anim = {}) {
                const a = parseInt(document.getElementById('fa').value, 10) || 2;
                const b = parseInt(document.getElementById('fb').value, 10) || 2;
                const svg = document.getElementById('frog-scene'), big = svg.getBoundingClientRect().width < 420;
                const L = frogLength(a, b, big), l = lcm(a, b), commons = [];
                for (let c = l; c <= L; c += l) commons.push(c);
                document.getElementById('fa-val').textContent = `по ${a}`;
                document.getElementById('fb-val').textContent = `по ${b}`;
                svg.innerHTML = frogsSvg(a, b, L, { A: frog.A, B: frog.B, common: frog.C, big, animA: anim.A, animB: anim.B, animC: anim.C });
                document.getElementById('frog-text').textContent = `Обидві приземлялися на ${commons.join(', ')}, … — це спільні кратні чисел ${a} і ${b}. Найменше з них — ${l}.`;
            }
            window.renderFrogs = function () {
                frog = { A: false, B: false, C: false };
                document.getElementById('frog-answer').classList.remove('shown');
                drawFrogs();
            };
            window.frogShow = function (k) {
                const was = { ...frog };
                if (k === 'A') frog.A = true;
                if (k === 'B') frog.B = true;
                if (k === 'C') frog = { A: true, B: true, C: true };
                drawFrogs({ A: frog.A && !was.A, B: frog.B && !was.B, C: frog.C && !was.C });
                if (k === 'C') document.getElementById('frog-answer').classList.add('shown');
            };

            // --- Слайд 4: кроки способів відкриваються по кліку ---
            window.nextStep = function (id) {
                const next = document.querySelector(`#${id} .step:not(.shown)`);
                if (next) next.classList.add('shown');
            };
            window.resetSteps = function (id) {
                document.querySelectorAll(`#${id} .step`).forEach((st) => st.classList.remove('shown'));
            };

            // --- Слайд 5: 12 — найменше спільне кратне, а 24 = 4 · 6 — ні ---
            window.renderTrap = function () {
                const svg = document.getElementById('trap-scene'), big = svg.getBoundingClientRect().width < 420, fs = big ? 21 : 15;
                const X = (m) => FX0 + m * FW / 28;
                svg.innerHTML = frogsSvg(4, 6, 28, { A: true, B: true, big, marks: [[12, 'good'], [24, 'bad']] })
                    + `<text x="${X(12)}" y="30" class="mark-text good" style="font-size:${fs}px">12 — найменше ✓</text>`
                    + `<text x="${X(24) + 16}" y="${YB + 62}" class="mark-text bad" style="font-size:${fs}px;text-anchor:end">4 · 6 = 24 — не найменше</text>`;
            };

            // Після зміни розміру вікна перемальовуємо доріжки (на вузькому екрані — більші підписи)
            window.addEventListener('resize', () => {
                const s = document.querySelector('.slide-card.active');
                if (s && s.dataset.slide === '2') drawFrogs();
                if (s && s.dataset.slide === '5') renderTrap();
            });
