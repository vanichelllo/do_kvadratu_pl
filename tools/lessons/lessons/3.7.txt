            // n рядів по k кружечків; перші m рядів зафарбовано; праворуч підписи рядів
            function dotsSvg(n, k, m, x0, y0, w, h, labels) {
                const rowH = h / n, step = Math.min(38, w / k), r = Math.min(13, step / 2 - 3, rowH / 2 - 4);
                let svg = '';
                for (let i = 0; i < n; i++) {
                    const y = y0 + i * rowH + rowH / 2;
                    svg += `<rect x="${x0 - 6}" y="${(y - rowH / 2 + 2).toFixed(1)}" width="${(k * step + 12).toFixed(1)}" height="${(rowH - 4).toFixed(1)}" rx="8" class="group-box"/>`;
                    for (let j = 0; j < k; j++) svg += `<circle cx="${(x0 + step / 2 + j * step).toFixed(1)}" cy="${y.toFixed(1)}" r="${r.toFixed(1)}" class="dot${i < m ? ' on' : ''}"/>`;
                    if (labels) svg += `<text x="${(x0 + k * step + 16).toFixed(1)}" y="${(y + 5).toFixed(1)}" class="row-lbl">1/${n} = ${k}</text>`;
                }
                return svg;
            }
            // Смужка: n рівних частин по v, перші m зафарбовано, у кожній частині підпис v
            function stripSvg(x, y, w, h, n, m, v) {
                let svg = '';
                for (let k = 0; k < n; k++) {
                    const cx = x + k * w / n;
                    svg += `<rect x="${cx.toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="strip-cell"/>`;
                    if (k < m) svg += `<rect x="${cx.toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${h}" class="strip-a" style="stroke:var(--text-dark);stroke-width:2"/>`;
                    svg += `<text x="${(cx + w / n / 2).toFixed(1)}" y="${y + h / 2 + 5}" class="svg-caption">${v}</text>`;
                }
                return svg;
            }

            function dotsScene(n, k, m, show) {
                return lbl(200, 22, `${m}/${n} від ${n * k} = ${show ? m * k : '?'}`, 'currentColor')

            }
