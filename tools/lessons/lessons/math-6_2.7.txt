            const typeset = (els) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise(els); };
            const fr = (a, b) => `\\dfrac{${a}}{${b}}`;
            // Горизонтальна смужка-ціле на k частин: зафарбовано m частин кольором color; labels — підписи частин
            function stripSvg(k, m, x0, y, w, h, color, labels = null) {
                const cw = w / k;
                let s = `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="cell empty"/>`;
                for (let i = 0; i < m; i++) s += `<rect x="${(x0 + i * cw).toFixed(1)}" y="${y}" width="${cw.toFixed(1)}" height="${h}" class="cell ${color}"/>`;
                for (let i = 1; i < k; i++) s += `<line x1="${(x0 + i * cw).toFixed(1)}" y1="${y}" x2="${(x0 + i * cw).toFixed(1)}" y2="${y + h}" class="cut-new"/>`;
                if (labels) for (let i = 0; i < k; i++) s += `<text x="${(x0 + (i + 0.5) * cw).toFixed(1)}" y="${y + h / 2 + 5}" class="piece-num">${labels}</text>`;
                return s + `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="frame"/>`;
            }
            // Фігурна дужка над (up) або під відрізком [x1; x2]
            function braceSvg(x1, x2, y, up, cls = '') {
                const d = up ? -10 : 10, xm = (x1 + x2) / 2;
                return `<path d="M${x1} ${y} q0 ${d} 10 ${d} H${xm - 8} q8 0 8 ${d} q0 ${-d} 8 ${-d} H${x2 - 10} q10 0 10 ${-d}" class="brace ${cls}"/>`;
            }

                s += `<rect x="${x}" y="${y0}" width="${w}" height="${3 * h}" rx="6" class="frame"/>`;
                // дужка праворуч від води: 2/3 = 40 л
                s += `<path d="M${x + w + 12} ${y0 + h} q10 0 10 10 V${y0 + 2 * h - 8} q0 8 8 8 q-8 0 -8 8 V${y0 + 3 * h - 10} q0 10 -10 10" class="brace"/>`;
