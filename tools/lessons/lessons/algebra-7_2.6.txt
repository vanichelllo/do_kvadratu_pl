            const typeset = (el) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([el]); };
            const sg = (n) => n < 0 ? `- ${-n}` : `+ ${n}`;
            const term = (k, v) => k === 0 ? '' : (k === 1 ? `+ ${v}` : k === -1 ? `- ${v}` : `${sg(k)}${v}`);
            const vals = () => [+document.getElementById('w-p').value, +document.getElementById('w-q').value];
            window.renderLab = function () {
                const [p, q] = vals();
                document.getElementById('w-p-val').textContent = p;
                document.getElementById('w-q-val').textContent = q;
                const head = (n) => n < 0 ? `-${-n}` : n;
                const t = document.getElementById('lab-table');
                t.innerHTML = `<table class="area"><tr><th></th><th>$x$</th><th>$${head(q)}$</th></tr>
                    <tr><th>$x$</th><td class="c1">?</td><td class="c2">?</td></tr>
                    <tr><th>$${head(p)}$</th><td class="c3">?</td><td class="c4">?</td></tr></table>`;
                const lq = document.getElementById('lab-q');
                lq.innerHTML = `$(x ${sg(p)})(x ${sg(q)}) = \\;?$`;
                document.getElementById('lab-a').classList.remove('shown');
                typeset(t); typeset(lq);
            };
            window.revealLab = function () {
                const [p, q] = vals(), head = (n) => n < 0 ? `-${-n}` : n, cx = (n) => n === 0 ? '0' : n === 1 ? 'x' : n === -1 ? '-x' : `${n}x`;
                const t = document.getElementById('lab-table');
                t.innerHTML = `<table class="area"><tr><th></th><th>$x$</th><th>$${head(q)}$</th></tr>
                    <tr><th>$x$</th><td class="c1">$x^2$</td><td class="c2">$${cx(q)}$</td></tr>
                    <tr><th>$${head(p)}$</th><td class="c3">$${cx(p)}$</td><td class="c4">$${p * q}$</td></tr></table>`;
                const a = document.getElementById('lab-a');
                const free = p * q === 0 ? '' : ` ${sg(p * q)}`;
                a.innerHTML = `$= x^2 ${term(p + q, 'x')}${free}$`.replace('$= x^2 $', '$= x^2$');
                a.classList.add('shown');
                typeset(t); typeset(a);
            };
