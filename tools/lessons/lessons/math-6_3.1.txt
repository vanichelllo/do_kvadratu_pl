            const typeset = (els) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise(els); };
            const gcd = (a, b) => b ? gcd(b, a % b) : a;
            // Склянка: x, y — лівий верхній кут; kind — syrup / water
            const cupSvg = (x, y, kind, pop = false) =>
                `<path d="M${x} ${y} L${x + 28} ${y} L${x + 24} ${y + 40} L${x + 4} ${y + 40} Z" class="cup ${kind}${pop ? ' pop' : ''}"/>`;
            // Ряд склянок: n штук, починаючи з x, крок step
            function cupsRow(n, x, y, kind, step = 34, popFrom = n) {
                let s = '';
                for (let i = 0; i < n; i++) s += cupSvg(x + i * step, y, kind, i >= popFrom);
                return s;
            }

            function drawMix(animate) {
                const [a, b, k] = mixVals();
                // по 9 склянок у ряд; сироп і вода окремими рядами
                const rows = (n, y0, kind) => {
                    let s = '';
                    for (let i = 0; i < n; i++) s += cupSvg(20 + (i % 9) * 44, y0 + Math.floor(i / 9) * 50, kind, animate && i >= n / k * (k - 1) && k > 1);
                    return s;
                };
                let s = `<text x="210" y="18" class="svg-small">сироп: ${a * k}</text>` + rows(a * k, 26, 'syrup');
                const yW = 26 + Math.ceil(a * k / 9) * 50 + 22;
                s += `<text x="210" y="${yW - 8}" class="svg-small">вода: ${b * k}</text>` + rows(b * k, yW, 'water');
                document.getElementById('mix-scene').innerHTML = s;
            }
            window.revealMix = function () {
                const [a, b, k] = mixVals(), g = gcd(a, b), out = document.getElementById('mix-text');
                const times = b % a === 0 ? `у ${b / a} ${b / a < 5 ? 'рази' : 'разів'}` : `у $\\dfrac{${b / g}}{${a / g}}$ раза`;
                out.innerHTML = `$${a * k} : ${b * k} = ${a / g} : ${b / g}$ — води ${b === a ? 'стільки ж, скільки сиропу' : times + (b > a ? ' більше' : '')}.<br>Порцій більше — відношення те саме.`;
                if (b < a) out.innerHTML = `$${a * k} : ${b * k} = ${a / g} : ${b / g}$ — сиропу більше, ніж води.<br>Порцій більше — відношення те саме.`;
                mixShown = true;
                document.getElementById('mix-answer').classList.add('shown');
                typeset([out]);
            };
