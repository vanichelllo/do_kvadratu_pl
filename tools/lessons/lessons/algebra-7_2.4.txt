            // --- Слайд 4: P ± Q ---
            const Q = [1, -2, 4];
            let op = 0, polyShown = false;
            // Многочлен ax² + bx + c у TeX
            function polyTex([a, b, c]) {
                const parts = [];
                [[a, 'x^2'], [b, 'x'], [c, '']].forEach(([k, v]) => {
                    if (k === 0) return;
                    const abs = Math.abs(k), body = v ? (abs === 1 ? v : abs + v) : String(abs);
                    parts.push(parts.length ? (k > 0 ? ` + ${body}` : ` - ${body}`) : (k > 0 ? body : `-${body}`));
                });
                return parts.length ? parts.join('') : '0';
            }
            const termTex = (k, v) => k === 0 ? '0' : v ? (k === 1 ? v : k === -1 ? `-${v}` : `${k}${v}`) : String(k);
            function polyVals() { return ['w-a', 'w-b', 'w-c'].map(id => parseInt(document.getElementById(id).value, 10)); }
            window.setOp = function (k) {
                op = k;
                [0, 1].forEach(j => document.getElementById('op-' + j).classList.toggle('on', j === k));
                renderPoly();
            };
            function polyTable(P, show) {
                const R = P.map((p, i) => op === 0 ? p + Q[i] : p - Q[i]), V = ['x^2', 'x', ''];
                const row = (name, arr, cls = '') => `<tr><th>${name}</th>${arr.map((k, i) => `<td${cls ? ` class="${cls}"` : ''}>$${termTex(k, V[i])}$</td>`).join('')}</tr>`;
                return '<tr><th></th><th>$x^2$</th><th>$x$</th><th>число</th></tr>' + row('$P$', P) + row(op === 0 ? '$+\\,Q$' : '$-\\,Q$', op === 0 ? Q : Q.map(q => -q))
                    + (show ? row('разом', R, 'res') : '<tr><th>разом</th><td>?</td><td>?</td><td>?</td></tr>');
            }
            window.renderPoly = function () {
                const P = polyVals();
                ['a', 'b', 'c'].forEach((k, i) => { document.getElementById(`w-${k}-val`).textContent = String(P[i]).replace('-', '−'); });
                polyShown = false;
                const task = document.getElementById('poly-task'), table = document.getElementById('poly-table');
                task.innerHTML = `$(${polyTex(P)}) ${op === 0 ? '+' : '-'} (${polyTex(Q)}) = \\,?$`;
                table.innerHTML = polyTable(P, false);
                document.getElementById('poly-result').innerHTML = '';
                document.getElementById('poly-note').innerHTML = op === 1 ? 'Мінус перед дужками: у рядку $-Q$ знаки всіх членів змінено.' : '';
                typeset(task); typeset(table); typeset(document.getElementById('poly-note'));
            };
            window.revealPoly = function () {
                const P = polyVals(), R = P.map((p, i) => op === 0 ? p + Q[i] : p - Q[i]);
                const table = document.getElementById('poly-table'), res = document.getElementById('poly-result');
                table.innerHTML = polyTable(P, true);
                res.innerHTML = `$= $ <span class="ok">$${polyTex(R)}$</span>`;
                typeset(table); typeset(res);
            };
