            const v = (id) => +document.getElementById(id).value;
            window.renderLab = function () {
                const a = v('w-a'), b = v('w-b'), k = v('w-k');
                ['a', 'b', 'k'].forEach(n => document.getElementById(`w-${n}-val`).textContent = v('w-' + n));
                const p = document.getElementById('lab-prop');
                p.innerHTML = `$${a} : ${b} = ${a * k} : ${b * k}$`;
                document.getElementById('lab-out').classList.remove('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([p]);
            };
            window.revealLab = function () {
                const a = v('w-a'), b = v('w-b'), k = v('w-k'), o = document.getElementById('lab-out');
                o.innerHTML = `Крайні: $${a} \\cdot ${b * k} = ${a * b * k}$<br>Середні: $${b} \\cdot ${a * k} = ${a * b * k}$<br>Однакові! ✓`;
                o.classList.add('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([o]);
            };
