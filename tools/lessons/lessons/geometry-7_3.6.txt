            const f1 = (v) => v.toFixed(1);
            const seg = (p, q, cls) => `<line x1="${f1(p[0])}" y1="${f1(p[1])}" x2="${f1(q[0])}" y2="${f1(q[1])}" class="${cls}"/>`;
            const poly = (P, cls) => `<polygon points="${P.map(p => p.map(f1).join(',')).join(' ')}" class="${cls}"/>`;
            const txt = (p, s, cls = 'v-name') => `<text x="${f1(p[0])}" y="${f1(p[1])}" class="${cls}">${s}</text>`;
            // Знак прямого кута при C між напрямками на P і Q
            function rmark(C, P, Q, d = 14) {
                const u = [(P[0] - C[0]) / Math.hypot(P[0] - C[0], P[1] - C[1]), (P[1] - C[1]) / Math.hypot(P[0] - C[0], P[1] - C[1])];
                const w = [(Q[0] - C[0]) / Math.hypot(Q[0] - C[0], Q[1] - C[1]), (Q[1] - C[1]) / Math.hypot(Q[0] - C[0], Q[1] - C[1])];
                const a = [C[0] + u[0] * d, C[1] + u[1] * d], b = [a[0] + w[0] * d, a[1] + w[1] * d], c = [C[0] + w[0] * d, C[1] + w[1] * d];
                return `<path d="M${f1(a[0])} ${f1(a[1])} L${f1(b[0])} ${f1(b[1])} L${f1(c[0])} ${f1(c[1])}" class="right-mark"/>`;
            }
            function angleArc(V, P, Q, r, cls) {
                const a1 = Math.atan2(P[1] - V[1], P[0] - V[0]), a2 = Math.atan2(Q[1] - V[1], Q[0] - V[0]);
                let d = a2 - a1;
                while (d > Math.PI) d -= 2 * Math.PI;
                while (d < -Math.PI) d += 2 * Math.PI;
                const s = [V[0] + r * Math.cos(a1), V[1] + r * Math.sin(a1)], e = [V[0] + r * Math.cos(a1 + d), V[1] + r * Math.sin(a1 + d)];
                return `<path d="M${f1(V[0])} ${f1(V[1])} L${f1(s[0])} ${f1(s[1])} A${r} ${r} 0 0 ${d > 0 ? 1 : 0} ${f1(e[0])} ${f1(e[1])} Z" class="angle-arc ${cls}"/>`;
            }
