            const f1 = (v) => v.toFixed(1);
            const polyPts = (cx, cy, R, n, rot) => Array.from({ length: n }, (_, k) => {
                const t = rot + 2 * Math.PI * k / n; return [cx + R * Math.cos(t), cy + R * Math.sin(t)];
            });
            const poly = (P, cls) => `<polygon points="${P.map(p => p.map(f1).join(',')).join(' ')}" class="${cls}"/>`;
            window.renderHex = function () {
                const R = 36, dx = R * 1.5, dy = R * Math.sqrt(3);
                let s = '';
                for (let i = 0; i < 6; i++) for (let j = 0; j < 4; j++) {
                    const cx = 50 + i * dx, cy = 40 + j * dy + (i % 2 ? dy / 2 : 0);
                    if (cy < 250) s += poly(polyPts(cx, cy, R, 6, 0), 'poly');
                }
                document.getElementById('hex-scene').innerHTML = s;
            };
            window.renderLab = function () {
                const n = +document.getElementById('w-n').value, R = 120, c = 150, rr = R * Math.cos(Math.PI / n), rot = -Math.PI / 2;
                document.getElementById('w-n-val').textContent = n;
                const P = polyPts(c, c, R, n, rot), M = [(P[0][0] + P[1][0]) / 2, (P[0][1] + P[1][1]) / 2];
                let s = `<circle cx="${c}" cy="${c}" r="${R}" class="circ-out"/><circle cx="${c}" cy="${c}" r="${f1(rr)}" class="circ-in"/>` + poly(P, 'poly');
                s += `<line x1="${c}" y1="${c}" x2="${f1(P[0][0])}" y2="${f1(P[0][1])}" class="rad"/><line x1="${c}" y1="${c}" x2="${f1(M[0])}" y2="${f1(M[1])}" class="rad-in"/>`;
