            // --- Лінійка: 0–12 см, 30 px на сантиметр (3 px на міліметр), нуль при x = 30 ---
            const R0 = 30, PX = 3;
            function rulerSvg(y, maxCm) {
                let svg = `<rect x="${R0 - 12}" y="${y}" width="${maxCm * 10 * PX + 24}" height="46" rx="4" class="ruler"/>`;
                for (let mm = 0; mm <= maxCm * 10; mm++) {
                    const x = R0 + mm * PX, h = mm % 10 === 0 ? 16 : mm % 5 === 0 ? 11 : 7;
                    svg += `<line x1="${x}" y1="${y}" x2="${x}" y2="${y + h}" class="tick${mm % 10 === 0 ? ' cm' : ''}"/>`;
                    if (mm % 10 === 0) svg += `<text x="${x}" y="${y + 32}" class="tick-num">${mm / 10}</text>`;
                }
                return svg;
            }
            function point(x, y, name) {
                return `<circle cx="${x}" cy="${y}" r="6" class="pt"/><text x="${x}" y="${y - 14}" class="pt-name">${name}</text>`;
            }
            function cmMm(mm) {
                const cm = Math.floor(mm / 10), r = mm % 10;
                return (cm ? `${cm} см` : '') + (cm && r ? ' ' : '') + (r ? `${r} мм` : '');
            }

            // --- Слайд 2: точка B рухається, лінійку можна «прикласти» анімацією ---
            window.renderRuler = function () {
                const b = parseInt(document.getElementById('w-b').value, 10) || 10;
                document.getElementById('w-b-val').textContent = cmMm(b);
                document.getElementById('ruler-answer').classList.remove('shown');
                document.getElementById('ruler-scene').innerHTML =
                    `<line x1="${R0}" y1="70" x2="${R0 + b * PX}" y2="70" class="seg"/>` + point(R0, 70, 'A') + point(R0 + b * PX, 70, 'B')
                    + `<line x1="${R0}" y1="70" x2="${R0}" y2="110" class="tick" stroke-dasharray="3 3"/>`
                    + `<line x1="${R0 + b * PX}" y1="70" x2="${R0 + b * PX}" y2="110" class="tick" stroke-dasharray="3 3"/>`
                    + `<g id="ruler-g" class="ruler-move">${rulerSvg(110, 12)}</g>`;
            };
            window.applyRuler = function () {
                renderRuler();
                const g = document.getElementById('ruler-g');
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) return;
                g.style.transition = 'none';
                g.style.transform = 'translate(160px, 60px)';
                requestAnimationFrame(() => requestAnimationFrame(() => { g.style.transition = ''; g.style.transform = 'translate(0px, 0px)'; }));
            };
            window.revealRuler = function () {
                const b = parseInt(document.getElementById('w-b').value, 10) || 10;
                document.getElementById('ruler-len').textContent = `AB = ${cmMm(b)} = ${b} мм`;
                document.getElementById('ruler-answer').classList.add('shown');
            };
