            // --- Слайд 2: m синіх і n жовтих брусочків «a» з'єднуються в один ряд «Разом» ---
            const BW = 28, STEP = 30, X0 = 60;
            function brick(col, y, cls) {
                return `<g class="bk"><rect x="${X0 + col * STEP}" y="${y}" width="${BW}" height="40" rx="4" class="brick ${cls}"/>`
                     + `<text x="${X0 + col * STEP + BW / 2}" y="${y + 26}" class="brick-t ${cls}">a</text></g>`;
            }
            function bricksVals() {
                return [parseInt(document.getElementById('w-m').value, 10) || 1, parseInt(document.getElementById('w-n').value, 10) || 1];
            }
            window.renderBricks = function () {
                const [m, n] = bricksVals();
                document.getElementById('w-m-val').textContent = m;
                document.getElementById('w-n-val').textContent = n;
                document.getElementById('bricks-answer').classList.remove('shown');
                let svg = `<text x="4" y="56" class="row-label">${m}a</text><text x="4" y="126" class="row-label">${n}a</text>`
                        + `<text x="4" y="196" class="row-label" style="font-size:13px">Разом</text>`;
                for (let i = 0; i < m; i++) svg += brick(i, 30, 'm');
                for (let j = 0; j < n; j++) svg += brick(j, 100, 'n');
                document.getElementById('bricks-scene').innerHTML = svg;
            };
            window.joinBricks = function () {
                const [m, n] = bricksVals();
                renderBricks();
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                const groups = document.querySelectorAll('#bricks-scene .bk');
                requestAnimationFrame(() => requestAnimationFrame(() => groups.forEach((g, k) => {
                    // сині опускаються на 140 px, жовті — на 70 px і стають у ряд після синіх
                    g.style.transform = k < m ? 'translate(0px, 140px)' : `translate(${m * STEP}px, 70px)`;
                    if (reduce) g.style.transition = 'none';
                })));
                document.getElementById('bricks-sum').textContent = `${m}a + ${n}a = ${m + n}a`;
                setTimeout(() => document.getElementById('bricks-answer').classList.add('shown'), reduce ? 0 : 950);
            };
