            const typeset = (els) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise(els); };
            const fr = (a, b) => `\\dfrac{${a}}{${b}}`;
            // Купки предметів: b купок по k кружечків (купка — рамка); перші a купок зафарбовано, якщо on
            function groupsSvg(b, k, a, on, x0, y0, W, H, vertical) {
                // vertical: купки — стовпчики (для слайда 1), інакше — рядки
                let s = '';
                const gw = vertical ? W / b : W, gh = vertical ? H : H / b;
                const r = Math.min(vertical ? gw / 2.6 : gh / 2.6, vertical ? gh / (k * 2.4) : gw / (k * 2.4), 16);
                for (let g = 0; g < b; g++) {
                    const gx = vertical ? x0 + g * gw : x0, gy = vertical ? y0 : y0 + g * gh, sel = on && g < a;
                    s += `<rect x="${(gx + 3).toFixed(1)}" y="${(gy + 3).toFixed(1)}" width="${(gw - 6).toFixed(1)}" height="${(gh - 6).toFixed(1)}" rx="8" class="group-box${sel ? ' on' : ''}"/>`;
                    for (let i = 0; i < k; i++) {
                        const cx = vertical ? gx + gw / 2 : gx + (i + 0.5) * gw / k, cy = vertical ? gy + (i + 0.5) * gh / k : gy + gh / 2;
                        s += `<circle cx="${cx.toFixed(1)}" cy="${cy.toFixed(1)}" r="${r.toFixed(1)}" class="${sel ? 'dot-on' : 'dot-off'}"/>`;
                    }
                }
                return s;
            }
