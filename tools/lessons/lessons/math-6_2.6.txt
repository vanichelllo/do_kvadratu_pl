            const typeset = (els) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise(els); };
            const fr = (a, b) => `\\dfrac{${a}}{${b}}`;
            // Смужка-ціле на k частин: зафарбовано m частин кольором color, нумерація з номера start (0 — без номерів)
            function stripSvg(k, m, x0, y, w, h, color, start = 0) {
                const cw = w / k;
                let s = `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="cell empty"/>`;
                for (let i = 0; i < m; i++) s += `<rect x="${(x0 + i * cw).toFixed(1)}" y="${y}" width="${cw.toFixed(1)}" height="${h}" class="cell ${start && i % 2 ? 'alt' : color}"/>`;
                for (let i = 1; i < k; i++) s += `<line x1="${(x0 + i * cw).toFixed(1)}" y1="${y}" x2="${(x0 + i * cw).toFixed(1)}" y2="${y + h}" class="cut-new"/>`;
                if (start) for (let i = 0; i < m; i++) s += `<text x="${(x0 + (i + 0.5) * cw).toFixed(1)}" y="${y + h / 2 + 5}" class="piece-num">${start + i}</text>`;
                return s + `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="frame"/>`;
            }
