            const typeset = (el) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([el]); };

            const pdot = (P, x, y, label, dx = 8, dy = 16) => `<circle cx="${P.sx(x)}" cy="${P.sy(y)}" r="4.5" class="pt"/>` + (label ? `<text x="${P.sx(x) + dx}" y="${P.sy(y) + dy}" class="plabel">${label}</text>` : '');

            function qVals() {
                const A = document.getElementById('w-a');
                if (parseFloat(A.value) === 0) A.value = 0.5;

            function qTex(a, b, c) {
                const ta = a === 1 ? '' : a === -1 ? '-' : fmtN(a).replace('−', '-').replace(',', '{,}');
                const tb = b === 0 ? '' : b === 1 ? ' + x' : b === -1 ? ' - x' : b > 0 ? ` + ${b}x` : ` - ${-b}x`;
                const tc = c === 0 ? '' : c > 0 ? ` + ${c}` : ` - ${-c}`;
                return `y = ${ta}x^2${tb}${tc}`;
            }
            function qScene() {
                const [a, b, c] = qVals(), f = (x) => a * x * x + b * x + c, x0 = -b / (2 * a), y0 = f(x0);
                let svg = axesSvg(PQ);
                if (qShown && x0 >= PQ.x0 && x0 <= PQ.x1) svg += symAxis(PQ, x0);
                svg += curveSvg(PQ, f, -6, 6, () => 'pos');
                if (qShown) {
                    svg += pdot(PQ, 0, c);
                    const D = b * b - 4 * a * c;
                    if (D >= 0) [(-b - Math.sqrt(D)) / (2 * a), (-b + Math.sqrt(D)) / (2 * a)].forEach(r => { if (r >= PQ.x0 && r <= PQ.x1) svg += pdot(PQ, r, 0); });
                    if (x0 >= PQ.x0 && x0 <= PQ.x1 && y0 >= PQ.y0 && y0 <= PQ.y1) svg += vdot(PQ, x0, y0, `(${fmtN(x0)}; ${fmtN(y0)})`);
                }
                return svg;
            }
            window.renderQ = function () {

            };
            window.revealQ = function () {
