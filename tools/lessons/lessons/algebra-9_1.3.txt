            // Нерівність → проміжок {lo, loClosed, hi, hiClosed}
            function ray(sign, a) {
                if (sign === '>' || sign === '≥') return { lo: a, loC: sign === '≥', hi: Infinity, hiC: false };
                return { lo: -Infinity, loC: false, hi: a, hiC: sign === '≤' };
            }
            function intersect(p, q) {
                const lo = Math.max(p.lo, q.lo), hi = Math.min(p.hi, q.hi);
                const loC = (p.lo === lo ? p.loC : true) && (q.lo === lo ? q.loC : true);
                const hiC = (p.hi === hi ? p.hiC : true) && (q.hi === hi ? q.hiC : true);
                if (lo > hi || (lo === hi && !(loC && hiC))) return null;
                return { lo, loC, hi, hiC };
            }
            function intervalTex(r) {
                if (r.lo === r.hi) return `x = ${r.lo}`;
                const left = r.lo === -Infinity ? '(-\\infty' : `${r.loC ? '[' : '('}${r.lo}`;
                const right = r.hi === Infinity ? '+\\infty)' : `${r.hi}${r.hiC ? ']' : ')'}`;
                return `x \\in ${left};\\,${right}`;
            }

                [[1, a, 60], [2, b, 105]].forEach(([k, v, y]) => {
                    const r = ray(signs[k], v), x1 = sx(r.lo), x2 = sx(r.hi);
                    svg += `<rect x="${x1}" y="${y}" width="${x2 - x1}" height="26" class="band-${k}"/>`
                         + `<line x1="${sx(v)}" y1="${y}" x2="${sx(v)}" y2="${Y}" class="band-edge-${k}" stroke-dasharray="4 4"/>`
                         + `<text x="${r.lo === -Infinity ? x1 + 6 : x2 - 6}" y="${y - 6}" class="band-label" style="fill:var(${k === 1 ? '--brand-blue' : '--brand-yellow'});text-anchor:${r.lo === -Infinity ? 'start' : 'end'}">${k}) x ${signs[k]} ${minus(v)}</text>`;
                });
                const r = intersect(ray(signs[1], a), ray(signs[2], b));
                svg += `<line x1="${sx(X0)}" y1="${Y}" x2="${sx(X1) + 6}" y2="${Y}" class="axis"/><path d="M${sx(X1) + 10} ${Y} l-10 -5 v10 z" fill="currentColor"/>`
                     + `<text x="${sx(X1) + 4}" y="${Y + 22}" class="axis-name">x</text>`;
                for (let x = -6; x <= 6; x++) svg += `<line x1="${sx(x)}" y1="${Y - 5}" x2="${sx(x)}" y2="${Y + 5}" class="tick-line"/><text x="${sx(x)}" y="${Y + 24}" class="tick">${minus(x)}</text>`;
                if (r && revealed) svg += r.lo === r.hi ? '' : `<line x1="${sx(r.lo)}" y1="${Y}" x2="${sx(r.hi)}" y2="${Y}" class="answer-band"/>`;
                [[a, ray(signs[1], a)], [b, ray(signs[2], b)]].forEach(([v, rr]) => {
                    const closed = rr.lo === v ? rr.loC : rr.hiC;
                    svg += `<circle cx="${sx(v)}" cy="${Y}" r="6" class="pt ${closed ? 'closed' : 'open'}"/>`;
                });
                if (r && revealed && r.lo === r.hi) svg += `<circle cx="${sx(r.lo)}" cy="${Y}" r="8" class="pt closed" style="fill:var(--brand-green);stroke:var(--brand-green)"/>`;
                document.getElementById('line-scene').innerHTML = svg;
                const text = r ? `$${intervalTex(r)}$` : 'Спільної частини немає — розв\'язків немає.';
                const out = document.getElementById('line-readout');
                out.innerHTML = `$\\begin{cases} x ${TEX[signs[1]]} ${a}, \\\\ x ${TEX[signs[2]]} ${b} \\end{cases}$ &nbsp;⇒&nbsp; ${text}`;
                if (revealed) document.getElementById('line-answer').classList.add('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([out]);
            };
            window.revealLine = function () { revealed = true; renderLine(); };
            window.lineInput = function () { revealed = false; renderLine(); };

            // --- Слайд 5: покроковий розв'язок системи ---
            const STEPS = [
                ['$3x - 2 \\gt 7 \\Rightarrow 3x \\gt 9 \\Rightarrow$ [[$x \\gt 3$]]', 'Перша нерівність'],
                ['$5 - x \\ge 1 \\Rightarrow -x \\ge -4 \\Rightarrow$ [[$x \\le 4$]]', 'Друга: ділимо на $-1$ — знак змінюється'],
                ['[[$3 \\lt x \\le 4$]]', 'Переріз променів $(3;\\,+\\infty)$ і $(-\\infty;\\,4]$'],
                ['Відповідь: $(3;\\,4]$', '$3$ не входить, $4$ входить'],
            ];
            let shown = 1;
            window.renderEval = function () {
                const box = document.getElementById('eval-lines');
                box.innerHTML = STEPS.slice(0, shown).map(([line], k) => {
                    const html = line.replace(/\[\[(.+?)\]\]/g, k === shown - 1 ? '<mark>$1</mark>' : '$1');
                    return `<div class="eval-line${k === STEPS.length - 1 ? ' done' : ''}">${html}</div>`;
                }).join('');
                document.getElementById('eval-note').innerHTML = STEPS[shown - 1][1];
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([box, document.getElementById('eval-note')]);
            };
            window.evalStep = function () { shown = Math.min(shown + 1, STEPS.length); renderEval(); };
            window.evalReset = function () { shown = 1; renderEval(); };

            // --- Слайд 6: кроки прикладів з'являються по кліку ---
            window.nextStep = function (id) {
                const next = document.querySelector(`#${id} .step:not(.shown)`);
                if (next) next.classList.add('shown');
            };

            // --- Слайд 8: перемикач завдань ---
            window.showTask = function (n) {
                [1, 2, 3].forEach((k) => {
                    document.getElementById(`task-${k}`).classList.toggle('on', k === n);
                    document.getElementById(`tab-${k}`).classList.toggle('on', k === n);
                });
            };
