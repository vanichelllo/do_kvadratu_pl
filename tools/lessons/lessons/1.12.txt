            // --- Слайд 2: квадрат n×n або куб n×n×n ---
            let shape = 'sq';
            window.setShape = function (s) {
                shape = s;
                document.getElementById('mode-sq').classList.toggle('on', s === 'sq');
                document.getElementById('mode-cube').classList.toggle('on', s === 'cube');
                const slider = document.getElementById('w-n');
                slider.max = s === 'sq' ? 10 : 6;
                if (parseInt(slider.value, 10) > slider.max) slider.value = slider.max;
                renderShape();
            };
            function cubeSvg(n, size, ox, oy) {
                // Ізометричний куб: передня грань, верх і бік — сітки n×n
                const c = size / n, dx = size * 0.45, dy = size * 0.35, cx = c * 0.45, cy = c * 0.35;
                let svg = '';
                for (let i = 0; i < n; i++) for (let j = 0; j < n; j++) {
                    svg += `<rect x="${ox + j * c}" y="${oy + i * c}" width="${c}" height="${c}" class="face-front"/>`;
                    const tx = ox + j * c + (n - 1 - i) * cx, ty = oy - (n - i) * cy;
                    svg += `<path d="M${tx} ${ty + cy} l${cx} ${-cy} h${c} l${-cx} ${cy} z" class="face-top"/>`;
                    const sx = ox + size + j * cx, sy = oy + i * c - j * cy;
                    svg += `<path d="M${sx} ${sy} l${cx} ${-cy} v${c} l${-cx} ${cy} z" class="face-side"/>`;
                }
                return svg;
            }
            window.renderShape = function () {
                const n = parseInt(document.getElementById('w-n').value, 10) || 1;
                document.getElementById('w-n-val').textContent = n;
                document.getElementById('shape-answer').classList.remove('shown');
                let svg = '';
                if (shape === 'sq') {
                    const c = Math.min(28, 270 / n), ox = (360 - n * c) / 2, oy = (300 - n * c) / 2;
                    for (let i = 0; i < n; i++) for (let j = 0; j < n; j++) svg += `<rect x="${ox + j * c}" y="${oy + i * c}" width="${c}" height="${c}" class="cell-sq"/>`;
                } else {
                    const size = 180, ox = (360 - size * 1.45) / 2, oy = 90;
                    svg = cubeSvg(n, size, ox, oy);
                }
                document.getElementById('shape-scene').innerHTML = svg;
            };
            window.revealShape = function () {
                const n = parseInt(document.getElementById('w-n').value, 10) || 1;
                document.getElementById('shape-count').textContent = shape === 'sq'
                    ? `${n}² = ${n} · ${n} = ${n * n} клітинок`
                    : `${n}³ = ${n} · ${n} · ${n} = ${n * n * n} кубиків`;
                document.getElementById('shape-answer').classList.add('shown');
            };

            // --- Слайд 3: таблиця квадратів (1–10) і кубів (1–5) ---
            window.renderTable = function () {
                let svg = '';
                const rows = [['n', (k) => k], ['n²', (k) => k * k], ['n³', (k) => (k <= 5 ? k * k * k : '')]];
                rows.forEach(([head, f], r) => {
                    const y = 50 + r * 70;
                    svg += `<text x="22" y="${y}" class="tab-head">${head}</text>`;
                    for (let k = 1; k <= 10; k++) svg += `<text x="${22 + k * 36}" y="${y}" class="tab-cell"${r === 0 ? ' style="font-weight:800"' : ''}>${f(k)}</text>`;
                    if (r < 2) svg += `<line x1="5" y1="${y + 30}" x2="395" y2="${y + 30}" class="tab-line"/>`;
                });
                svg += `<text x="200" y="240" class="tab-cell" style="font-size:13px">Корисно пам'ятати квадрати чисел до 10 і куби до 5</text>`;
                document.getElementById('table-scene').innerHTML = svg;
            };

            let exIndex = 0, shownLines = 1;

            window.renderEval = function () {

                    return `<div class="eval-line${k === ex.lines.length - 1 ? ' done' : ''}">${k ? '= ' : ''}${html}</div>`;
                }).join('');
                document.getElementById('eval-note').textContent = ex.lines[shownLines - 1][1];
            };

            window.evalNext = function () { exIndex = (exIndex + 1) % EXAMPLES.length; shownLines = 1; renderEval(); };
