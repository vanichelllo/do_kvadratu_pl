            // --- Спільне: MathJax, дріб у SVG і смужка з частинами ---
                        // Дріб у SVG: чисельник над рискою (MathJax у <text> не працює); y — рівень риски
            function fracSvg(x, y, a, b, color, size = 22) {
                const fill = color ? `;fill:${color}` : '', stroke = color ? ` style="stroke:${color}"` : '';
                const half = String(Math.max(a, b)).length * size * 0.3 + 5;
                return `<text x="${x}" y="${y - size * 0.3}" class="fr-t" style="font-size:${size}px${fill}">${a}</text>`
                     + `<line x1="${x - half}" y1="${y}" x2="${x + half}" y2="${y}" class="fr-bar"${stroke}/>`
                     + `<text x="${x}" y="${y + size * 1.05}" class="fr-t" style="font-size:${size}px${fill}">${b}</text>`;
            }
            // Смужка з b частин, кожну з яких поділено ще на k; зафарбовано m дрібних частин
            function stripSvg(m, b, k, x0, y, w, h, color) {
                const n = b * k, cw = w / n;
                let s = `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="cell empty"/>`
                      + `<rect x="${x0}" y="${y}" width="${m * cw}" height="${h}" class="cell ${color}"/>`;
                for (let i = 1; i < n; i++) {
                    const x = x0 + i * cw;
                    s += i % k ? `<line x1="${x}" y1="${y}" x2="${x}" y2="${y + h}" class="cut-new"/>`
                               : `<line x1="${x}" y1="${y - 6}" x2="${x}" y2="${y + h + 6}" class="cut-main"/>`;
                }
                return s + `<rect x="${x0}" y="${y}" width="${w}" height="${h}" class="frame"/>`;
            }

            const typeset = (els) => { if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise(els); };
            const gcd = (a, b) => (b ? gcd(b, a % b) : a);
            const fr = (a, b) => `\\dfrac{${a}}{${b}}`;
            // Модель площі: ціле — прямокутник; b стовпчиків (a синіх), d рядків (c зелених)
            function areaSvg(a, b, c, d, x0, y0, W, H, showRows) {
                const cw = W / b, rh = H / d;
                let s = '';
                for (let i = 0; i < b; i++) for (let j = 0; j < d; j++) {
                    const x = x0 + i * cw, y = y0 + j * rh;
                    s += `<rect x="${x.toFixed(1)}" y="${y.toFixed(1)}" width="${cw.toFixed(1)}" height="${rh.toFixed(1)}" class="grid-cell"/>`;
                    if (i < a) s += `<rect x="${x.toFixed(1)}" y="${y.toFixed(1)}" width="${cw.toFixed(1)}" height="${rh.toFixed(1)}" class="col-on"/>`;
                    if (showRows && i < a && j < c) s += `<rect x="${x.toFixed(1)}" y="${y.toFixed(1)}" width="${cw.toFixed(1)}" height="${rh.toFixed(1)}" class="row-on"/>`;
                }
                return s + `<rect x="${x0}" y="${y0}" width="${W}" height="${H}" class="frame"/>`;
            }
