            // --- Спільне: точки й стрілки ---
            const A = { x: 140, y: 140 }, B = { x: 270, y: 90 };
            const W = 420, H = 220, PAD = 14;
            function edgeT(P, Q, sign) {
                // Найбільше t (за знаком sign), при якому P + t(Q − P) ще в межах рисунка
                const dx = Q.x - P.x, dy = Q.y - P.y;
                let t = Infinity;
                if (dx) t = Math.min(t, ((sign * dx > 0 ? W - PAD : PAD) - P.x) / (sign * dx));
                if (dy) t = Math.min(t, ((sign * dy > 0 ? H - PAD : PAD) - P.y) / (sign * dy));
                return sign * t;
            }
            function at(P, Q, t) { return { x: P.x + t * (Q.x - P.x), y: P.y + t * (Q.y - P.y) }; }
            function arrow(from, to, cls) {
                const ang = Math.atan2(to.y - from.y, to.x - from.x), s = 13;
                const p1 = { x: to.x - s * Math.cos(ang - 0.4), y: to.y - s * Math.sin(ang - 0.4) };
                const p2 = { x: to.x - s * Math.cos(ang + 0.4), y: to.y - s * Math.sin(ang + 0.4) };
                return `<path d="M${to.x} ${to.y} L${p1.x} ${p1.y} L${p2.x} ${p2.y} Z" class="arrow-head ${cls || ''}"/>`;
            }
            function pt(P, name, start) {
                return `<circle cx="${P.x}" cy="${P.y}" r="6" class="pt${start ? ' start' : ''}"/><text x="${P.x}" y="${P.y - 14}" class="pt-name">${name}</text>`;
            }

            function figureSvg(t0, t1) {
                const p0 = at(A, B, t0), p1 = at(A, B, t1), f = FIG[figMode];
                let svg = `<line x1="${at(A, B, edgeT(A, B, -1)).x}" y1="${at(A, B, edgeT(A, B, -1)).y}" x2="${at(A, B, edgeT(A, B, 1)).x}" y2="${at(A, B, edgeT(A, B, 1)).y}" class="fig-line ghost"/>`;
                svg += `<line x1="${p0.x}" y1="${p0.y}" x2="${p1.x}" y2="${p1.y}" class="fig-line"/>`;
                const tMax = edgeT(A, B, 1), tMin = edgeT(A, B, -1);
                if ((figMode === 'ray' || figMode === 'line') && t1 >= tMax - 0.001) svg += arrow(at(A, B, 0), p1);
                if ((figMode === 'rayba' || figMode === 'line') && t0 <= tMin + 0.001) svg += arrow(at(A, B, 1), p0);
                svg += pt(A, 'A', figMode === 'ray') + pt(B, 'B', figMode === 'rayba');
                if (figMode === 'ray' || figMode === 'rayba') svg += `<text x="210" y="210" class="svg-caption">жовта точка — початок</text>`;
                return svg;
            }
            // Фігура «виростає» з відрізка AB до країв рисунка (кнопки запускає вчитель)
            function animateFigure() {
                cancelAnimationFrame(figAnim);
                const f = FIG[figMode], tMax = edgeT(A, B, 1), tMin = edgeT(A, B, -1);
                const end0 = f.from === null || f.back ? tMin : f.from, end1 = f.to === null && !f.back ? tMax : 1;
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                const scene = document.getElementById('figure-scene');
                if (reduce) { scene.innerHTML = figureSvg(end0, end1); return; }
                const t0 = performance.now(), dur = 900;
                const step = (now) => {
                    const k = Math.min(1, (now - t0) / dur), e = 1 - Math.pow(1 - k, 3);
                    scene.innerHTML = figureSvg(0 + (end0 - 0) * e, 1 + (end1 - 1) * e);
                    if (k < 1) figAnim = requestAnimationFrame(step);
                };
                figAnim = requestAnimationFrame(step);
            }
