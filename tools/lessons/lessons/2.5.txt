            // --- Слайд 2: правильний многокутник з n вершинами й «бігунець» по сторонах ---
            const NAMES = { 3: 'трикутник', 4: 'чотирикутник', 5: "п'ятикутник", 6: 'шестикутник', 7: 'семикутник', 8: 'восьмикутник' };
            const LETTERS = 'ABCDEFGH';
            function polyPoints(n) {
                const cx = 210, cy = 118, r = 85, pts = [];
                for (let k = 0; k < n; k++) {
                    const a = -Math.PI / 2 + (2 * Math.PI * k) / n;
                    pts.push({ x: cx + r * Math.cos(a), y: cy + r * Math.sin(a) });
                }
                return pts;
            }
            function polySvg(n, runner) {
                const pts = polyPoints(n), cx = 210, cy = 118;
                let svg = `<polygon points="${pts.map((p) => `${p.x},${p.y}`).join(' ')}" class="poly"/>`;
                pts.forEach((p, k) => {
                    const lx = cx + (p.x - cx) * 1.2, ly = cy + (p.y - cy) * 1.2 + 6;
                    svg += `<circle cx="${p.x}" cy="${p.y}" r="5" class="vx"/><text x="${lx}" y="${ly}" class="pt-name">${LETTERS[k]}</text>`;
                });
                if (runner) svg += `<circle cx="${runner.x}" cy="${runner.y}" r="9" class="runner"/>`;
                return svg;
            }
            function polyN() { return parseInt(document.getElementById('w-n').value, 10) || 3; }
            let polyAnim = null;
            window.renderPoly = function () {
                cancelAnimationFrame(polyAnim);
                const n = polyN();
                document.getElementById('w-n-val').textContent = n;
                document.getElementById('poly-answer').classList.remove('shown');
                document.getElementById('poly-scene').innerHTML = polySvg(n);
            };
            // Бігунець обходить замкнену ламану й повертається у вершину A
            window.walkPoly = function () {
                renderPoly();
                const n = polyN(), pts = polyPoints(n);
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) return;
                const t0 = performance.now(), dur = 600 * n;
                const step = (now) => {
                    const k = Math.min(1, Math.max(0, (now - t0) / dur)), pos = k * n, i = Math.min(n - 1, Math.floor(pos)), f = pos - i;
                    const a = pts[i], b = pts[(i + 1) % n];
                    document.getElementById('poly-scene').innerHTML = polySvg(n, { x: a.x + (b.x - a.x) * f, y: a.y + (b.y - a.y) * f });
                    if (k < 1) polyAnim = requestAnimationFrame(step);
                };
                polyAnim = requestAnimationFrame(step);
            };
            window.revealPoly = function () {
                const n = polyN();
                const few = n <= 4;   // 3–4 вершини, 5–8 вершин
                document.getElementById('poly-text').textContent = `${NAMES[n][0].toUpperCase() + NAMES[n].slice(1)} ${LETTERS.slice(0, n)}: ${n} ${few ? 'вершини' : 'вершин'}, `
                    + `${n} ${few ? 'сторони' : 'сторін'}, ${n} ${few ? 'кути' : 'кутів'}`;
                document.getElementById('poly-answer').classList.add('shown');
            };

            // --- Слайд 4: вид трикутника KLM ---
            const STEPS = [
                ['Найбільший кут — [[$106°$]].', 'Дивимось на найбільший кут'],
                ['$106°$ — тупий, отже, трикутник [[тупокутний]].', 'Один тупий кут — уже тупокутний'],
                ['$KL = KM = 5$ см — [[дві сторони рівні]].', 'Рисочки на малюнку теж показують рівні сторони'],
                ['Отже, трикутник [[рівнобедрений]].', 'Третя сторона, $8$ см, інша'],
                ['Відповідь: тупокутний рівнобедрений трикутник.', 'Готово!'],
            ];
