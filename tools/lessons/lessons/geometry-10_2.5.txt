            // --- Слайд 5: оригінал → зображення (афінне перетворення, як при паралельному проектуванні) ---
            function fitPts(P, x0, x1, y0, y1) {
                const xs = P.map((p) => p[0]), ys = P.map((p) => p[1]);
                const mnx = Math.min(...xs), mxx = Math.max(...xs), mny = Math.min(...ys), mxy = Math.max(...ys);
                const k = Math.min((x1 - x0) / (mxx - mnx), (y1 - y0) / (mxy - mny));
                const ox = (x0 + x1) / 2 - k * (mnx + mxx) / 2, oy = (y0 + y1) / 2 + k * (mny + mxy) / 2;
                return (p) => [r1(ox + k * p[0]), r1(oy - k * p[1])];
            }

            // --- Слайд 6: куб, призма, піраміда; невидимі ребра — штрихові ---
            function solidSvg(V, F, k, u0, v0) {
                const q = 0.5, a = 45 * RAD;
                const proj = ([x, y, z]) => [r1(u0 + k * (x + q * y * Math.cos(a))), r1(v0 - k * (z + q * y * Math.sin(a)))];
                const dir = [-q * Math.cos(a), 1, -q * Math.sin(a)];   // цей напрям рисунок «стискає» в точку
                const c = V.reduce((acc, p) => acc.map((t, i) => t + p[i] / V.length), [0, 0, 0]);
                const front = F.map((f) => {
                    const p = V[f[0]], u = V[f[1]].map((t, i) => t - p[i]), w = V[f[2]].map((t, i) => t - p[i]);
                    let n = [u[1] * w[2] - u[2] * w[1], u[2] * w[0] - u[0] * w[2], u[0] * w[1] - u[1] * w[0]];
                    if (n.reduce((s, t, i) => s + t * (p[i] - c[i]), 0) < 0) n = n.map((t) => -t);
                    return n.reduce((s, t, i) => s + t * dir[i], 0) < 0;
                });
                const edges = new Map();
                F.forEach((f, fi) => f.forEach((v, j) => {
                    const w = f[(j + 1) % f.length], key = v < w ? `${v}-${w}` : `${w}-${v}`;
                    edges.set(key, edges.get(key) === true || front[fi]);
                }));
                let hid = '', vis = '';
                edges.forEach((visible, key) => {
                    const [i, j] = key.split('-').map(Number), ln = sLine(proj(V[i]), proj(V[j]), visible ? 'edge' : 'edge hid');
                    if (visible) vis += ln; else hid += ln;
                });
                return { hid, vis, proj };
            }
