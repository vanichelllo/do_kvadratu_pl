            const f1 = (v) => v.toFixed(1);
            const ln = (p, q, cls) => `<line x1="${f1(p[0])}" y1="${f1(p[1])}" x2="${f1(q[0])}" y2="${f1(q[1])}" class="${cls}"/>`;
            const lerp = (p, q, t) => [p[0] + (q[0] - p[0]) * t, p[1] + (q[1] - p[1]) * t];
            window.renderDiv = function () {
                const A = [30, 200], B = [390, 200], R = [330, 40], n = 7;
                let s = ln(A, B, 'side') + ln(A, R, 'side');
                const C = (k) => lerp(A, R, k / n);
                for (let k = 1; k <= n; k++) {
                    const c = C(k), b = lerp(A, B, k / n);
                    s += `<circle cx="${f1(c[0])}" cy="${f1(c[1])}" r="3.5" class="dot"/>`;
                    s += ln(c, b, 'par') + `<circle cx="${f1(b[0])}" cy="${f1(b[1])}" r="3.5" class="dot"/>`;
                }
                s += `<text x="${A[0] - 10}" y="${A[1] + 20}" class="v-name">A</text><text x="${B[0]}" y="${B[1] + 20}" class="v-name">B</text>`;
                document.getElementById('div-scene').innerHTML = s;
            };
            window.renderTh = function () {
