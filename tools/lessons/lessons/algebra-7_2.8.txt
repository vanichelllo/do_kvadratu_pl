            const POLYS = [
                ['$3a + 3b + ax + bx$', ['Групи: $(3a + 3b) + (ax + bx)$', '$= 3(a + b) + x(a + b)$', '$= (a + b)(3 + x)$']],
                ['$x^2 + xy - 4x - 4y$', ['Групи: $(x^2 + xy) - (4x + 4y)$', '$= x(x + y) - 4(x + y)$', '$= (x + y)(x - 4)$']],
                ['$6ab - 3a + 4b - 2$', ['Групи: $(6ab - 3a) + (4b - 2)$', '$= 3a(2b - 1) + 2(2b - 1)$', '$= (2b - 1)(3a + 2)$']],
            ];
            let polyK = 0, polyN = 0;
            window.pickPoly = function (k) {
                polyK = k; polyN = 0;
                POLYS.forEach((_, i) => document.getElementById('pb-' + i).classList.toggle('on', i === k));
                renderPoly();
            };
            window.polyStep = function () { polyN = Math.min(polyN + 1, POLYS[polyK][1].length); renderPoly(); };
            window.renderPoly = function () {
                const q = document.getElementById('poly-q'), box = document.getElementById('poly-steps');
                q.innerHTML = POLYS[polyK][0];
                document.getElementById('pb-' + polyK).classList.add('on');
                box.innerHTML = POLYS[polyK][1].slice(0, polyN).map((t, k) => `<div class="eval-line${k === 2 ? ' done' : ''}">${t}</div>`).join('') || '<p class="eval-note">Які члени об\'єднаємо?</p>';
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([q, box]);
            };
