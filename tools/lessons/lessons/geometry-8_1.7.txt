            const f1 = (v) => v.toFixed(1), R = 115, O = [200, 150];
            const P = (deg) => [O[0] + R * Math.cos(deg * Math.PI / 180), O[1] - R * Math.sin(deg * Math.PI / 180)];
            const poly = (Q, cls) => `<polygon points="${Q.map(p => p.map(f1).join(',')).join(' ')}" class="${cls}"/>`;
            const lbl = (p, s) => { const dx = p[0] - O[0], dy = p[1] - O[1], L = Math.hypot(dx, dy) || 1; return `<text x="${f1(p[0] + dx / L * 20)}" y="${f1(p[1] + dy / L * 20 + 6)}" class="v-name">${s}</text>`; };
            function angleArc(V, A, B, r, cls) {
                const a1 = Math.atan2(A[1] - V[1], A[0] - V[0]), a2 = Math.atan2(B[1] - V[1], B[0] - V[0]);
                let d = a2 - a1;
                while (d > Math.PI) d -= 2 * Math.PI;
                while (d < -Math.PI) d += 2 * Math.PI;
                const s = [V[0] + r * Math.cos(a1), V[1] + r * Math.sin(a1)], e = [V[0] + r * Math.cos(a1 + d), V[1] + r * Math.sin(a1 + d)];
                return `<path d="M${f1(V[0])} ${f1(V[1])} L${f1(s[0])} ${f1(s[1])} A${r} ${r} 0 0 ${d > 0 ? 1 : 0} ${f1(e[0])} ${f1(e[1])} Z" class="angle-arc ${cls}"/>`;
            }
            const ang = (V, A, B) => { const a = [A[0] - V[0], A[1] - V[1]], b = [B[0] - V[0], B[1] - V[1]]; return Math.acos((a[0] * b[0] + a[1] * b[1]) / Math.hypot(...a) / Math.hypot(...b)) * 180 / Math.PI; };
