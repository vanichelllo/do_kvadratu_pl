            // Штрихування; у кожного SVG свій id, бо візерунок із прихованого слайда не малюється
            const hatch = (id) => `<defs><pattern id="${id}" width="10" height="10" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="10" class="hatch-line"/></pattern></defs>`;
            // Смужки по n частин; parts[k] — клас k-ї частини ('' — порожня). perRow смужок у ряду.
            function barsSvg(n, parts, x0, y0, colW, rowH, perRow, w, h, hid) {
                const bars = Math.max(1, Math.ceil(parts.length / n));
                let svg = '';
                for (let i = 0; i < bars; i++) {
                    const x = x0 + (i % perRow) * colW, y = y0 + Math.floor(i / perRow) * rowH;
                    for (let k = 0; k < n; k++) {
                        const cx = (x + k * w / n).toFixed(1), cw = (w / n).toFixed(1), cls = parts[i * n + k] || '';
                        svg += `<rect x="${cx}" y="${y}" width="${cw}" height="${h}" class="strip-cell"/>`;
                        if (cls) svg += `<rect x="${cx}" y="${y}" width="${cw}" height="${h}" class="${cls}" style="stroke:var(--text-dark);stroke-width:2${cls === 'strip-gone' ? `;fill:url(#${hid})` : ''}"/>`;
                    }
                }
                return svg;
            }
            const fill = (count, cls) => Array(count).fill(cls);

            function mixText(m, n) {
                const q = Math.floor(m / n), r = m % n;
                return r === 0 ? String(q) : (q === 0 ? `${r}/${n}` : `${q} ${r}/${n}`);
            }
            function mixTex(q, r, n) {

            }
