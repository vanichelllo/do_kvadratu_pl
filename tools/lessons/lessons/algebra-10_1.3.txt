            let rootShown = false;
            window.rootInput = function () { rootShown = false; renderRoot(); };
            window.revealRoot = function () { rootShown = true; renderRoot(); };
            window.renderRoot = function () {
                const n = parseInt(document.getElementById('r-n').value, 10), x = parseFloat(document.getElementById('r-x').value);
                document.getElementById('r-n-val').textContent = n;
                document.getElementById('r-x-val').textContent = num(x);
                const even = n % 2 === 0, y = even ? Math.abs(x) : x;
                const P = plane(-5, 5, -4.5, 4.5);
                let svg = axesSvg(P);
                svg += curveSvg(P, (t) => t, -5, 5, () => 'ghost');
                if (rootShown) {
                    svg += curveSvg(P, (t) => (even ? Math.abs(t) : t), -4.5, 4.5);
                    svg += `<line x1="${P.sx(x)}" y1="${P.sy(0)}" x2="${P.sx(x)}" y2="${P.sy(y)}" class="probe-line"/><circle cx="${P.sx(x)}" cy="${P.sy(y)}" r="7" class="probe"/>`
                         + `<text x="${P.sx(-4.7)}" y="${P.sy(0.6)}" class="fact-label" style="fill:var(--brand-blue)">${even ? 'y = |x|' : 'y = x'}</text>`;
                } else {
                    svg += `<circle cx="${P.sx(x)}" cy="${P.sy(0)}" r="7" class="probe"/>`;
                }
                svg += `<text x="${P.sx(4.8)}" y="${P.sy(-4.1)}" class="fact-label" style="fill:var(--text-muted);text-anchor:end">пунктир: y = x</text>`;
                document.getElementById('root-scene').innerHTML = svg;
                const box = document.getElementById('root-answer'), out = document.getElementById('root-readout');
                box.classList.toggle('shown', rootShown);
                if (!rootShown) return;
                const xs = Number.isInteger(x) ? `${x}` : `${x}`.replace('.', '{,}');
                const pow = Math.pow(x, n), powS = Number.isInteger(pow) ? `${pow}` : `${pow}`.replace('.', '{,}');
                const ys = Number.isInteger(y) ? `${y}` : `${y}`.replace('.', '{,}');
                const rt = (v) => (n === 2 ? `\\sqrt{${v}}` : `\\sqrt[${n}]{${v}}`);
                out.innerHTML = `$${rt(`(${xs})^{${n}}`)} = ${rt(powS)} = ${ys}$`
                    + `<br>${even ? (x < 0 ? 'Парне $n$, $x \\lt 0$: результат — $|x|$, а не $x$.' : 'Парне $n$, $x \\ge 0$: $|x| = x$.') : 'Непарне $n$: результат — саме $x$, знак зберігається.'}`;
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([out]);
