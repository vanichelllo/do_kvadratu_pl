            // Смужка з n частин: перші m зелені, решта — жовті (якщо rest) або порожні; vals — підписи частин
            function stripSvg(x, y, w, h, n, m, rest, vals) {
                let svg = '';
                for (let k = 0; k < n; k++) {
                    const cx = x + k * w / n, cw = w / n;
                    svg += `<rect x="${cx.toFixed(1)}" y="${y}" width="${cw.toFixed(1)}" height="${h}" class="strip-cell"/>`;
                    const cls = k < m ? 'strip-a' : (rest ? 'strip-q' : '');
                    if (cls) svg += `<rect x="${cx.toFixed(1)}" y="${y}" width="${cw.toFixed(1)}" height="${h}" class="${cls}" style="stroke:var(--text-dark);stroke-width:2"/>`;
                    if (vals) svg += `<text x="${(cx + cw / 2).toFixed(1)}" y="${y + h / 2 + 5}" class="svg-caption">${vals}</text>`;
                }
                return svg;
            }
            // Фігурна дужка над відрізком [x1; x2] (up) або під ним, з підписом
            function brace(x1, x2, y, up, text, col) {
                const d = up ? -10 : 10, mid = (x1 + x2) / 2;
                return `<path d="M${x1} ${y} q0 ${d} 10 ${d} L${mid - 8} ${y + d} q8 0 8 ${d} q0 ${-d} 8 ${-d} L${x2 - 10} ${y + d} q10 0 10 ${-d}" class="brace" style="stroke:${col}"/>`
                    + `<text x="${mid}" y="${up ? y + 2 * d - 6 : y + 2 * d + 16}" class="svg-caption" style="fill:${col};font-size:17px">${text}</text>`;
            }
