            // --- Слайд 2: розкладаємо цукерки на рівні групи (лише дільники, щоб ділилось без остачі) ---
            let total = 24;
            const divisorsOf = (n) => Array.from({ length: n }, (_, i) => i + 1).filter(d => n % d === 0);
            window.setTotal = function (n) {
                total = n;
                const slider = document.getElementById('w-g');
                slider.max = divisorsOf(n).length - 1;
                slider.value = Math.min(parseInt(slider.value, 10) || 0, slider.max);
                renderGroups();
            };
            window.renderGroups = function () {
                const divs = divisorsOf(total);
                const groups = divs[Math.min(parseInt(document.getElementById('w-g').value, 10) || 0, divs.length - 1)];
                const per = total / groups;
                document.getElementById('w-g-val').textContent = groups;
                document.getElementById('group-answer').classList.remove('shown');
                // Групи — рядами по кілька штук; кожна група в пунктирній рамці
                const cols = groups <= 4 ? groups : Math.ceil(Math.sqrt(groups * 1.4));
                const rows = Math.ceil(groups / cols);
                const gw = 380 / cols, gh = 280 / rows;
                const inRow = Math.ceil(Math.sqrt(per * gw / gh));
                const inCol = Math.ceil(per / inRow);
                const r = Math.max(3, Math.min(14, (gw - 14) / inRow / 2.4, (gh - 14) / inCol / 2.4));
                let svg = '';
                for (let g = 0; g < groups; g++) {
                    const gx = 10 + (g % cols) * gw, gy = 10 + Math.floor(g / cols) * gh;
                    svg += `<rect x="${gx + 3}" y="${gy + 3}" width="${gw - 6}" height="${gh - 6}" rx="10" class="group-box"/>`;
                    for (let k = 0; k < per; k++) {
                        const cx = gx + gw / 2 + ((k % inRow) - (inRow - 1) / 2) * r * 2.4;
                        const cy = gy + gh / 2 + (Math.floor(k / inRow) - (inCol - 1) / 2) * r * 2.4;
                        svg += `<circle cx="${cx}" cy="${cy}" r="${r}" class="chip g${g % 3}"/>`;
                    }
                }
                document.getElementById('group-scene').innerHTML = svg;
            };
            window.revealGroups = function () {
                const divs = divisorsOf(total);
                const groups = divs[Math.min(parseInt(document.getElementById('w-g').value, 10) || 0, divs.length - 1)];
                document.getElementById('group-div').textContent = `${total} : ${groups} = ${total / groups}`;
                document.getElementById('group-check').textContent = `Перевірка: ${total / groups} · ${groups} = ${total} ✓`;
                document.getElementById('group-answer').classList.add('shown');
            };

            // --- Слайд 4: ділення кутом ---
            const PAIRS = [[936, 4], [816, 8], [1512, 36]];
            let pairIndex = 0, shownSteps = 0;
            function divSteps(n, d) {
                // Кожен крок: де закінчується неповне ділене, його значення, цифра частки, добуток, остача
                const D = String(n).split('').map(Number);
                let e = 0, cur = D[0];
                while (cur < d && e < D.length - 1) { e++; cur = cur * 10 + D[e]; }
                const steps = [];
                for (;;) {
                    const q = Math.floor(cur / d), prod = q * d, rem = cur - prod;
                    steps.push({ end: e, cur, q, prod, rem });
                    if (e === D.length - 1) break;
                    e++;
                    cur = rem * 10 + D[e];
                }
                return steps;
            }
            function stepText(s, k, steps, n, d) {
                if (k === 0) return `Перше неповне ділене — ${s.cur}, тож у частці буде ${steps.length} цифр${steps.length < 5 ? 'и' : ''}. ${s.cur} : ${d} = ${s.q}; ${s.q} · ${d} = ${s.prod}; остача ${s.rem}.`;
                if (s.q === 0) return `Зносимо цифру: ${s.cur} менше за ${d} — у частці пишемо 0.`;
                return `Зносимо цифру: ${s.cur} : ${d} = ${s.q}; ${s.q} · ${d} = ${s.prod}; остача ${s.rem}.`;
            }
            window.renderLongDivision = function () {
                const [n, d] = PAIRS[pairIndex];
                const steps = divSteps(n, d), shown = steps.slice(0, shownSteps);
                const D = String(n), w = 28, x0 = 40, rowH = 30;
                const cx = (i) => x0 + i * w + w / 2;   // центр цифри з номером i (зліва)
                const right = (str, end, y, cls) => str.split('').map((ch, k) =>
                    `<text x="${cx(end - (str.length - 1 - k))}" y="${y}" class="${cls}">${ch}</text>`).join('');
                const xBar = x0 + D.length * w + 8;
                let svg = right(D, D.length - 1, 30, 'ld-digit')
                        + `<line x1="${xBar}" y1="8" x2="${xBar}" y2="72" class="ld-line"/>`
                        + `<text x="${xBar + 12}" y="30" class="ld-digit" style="text-anchor:start">${d}</text>`
                        + `<line x1="${xBar}" y1="40" x2="${xBar + 90}" y2="40" class="ld-line"/>`;
                const quot = shown.map(s => s.q).join('');
                quot.split('').forEach((ch, k) => {
                    svg += `<text x="${xBar + 12 + k * 16}" y="66" class="ld-quot${ch === '0' && k > 0 ? ' ld-zero' : ''}" style="text-anchor:start">${ch}</text>`;
                });
                let y = 30;
                shown.forEach((s, k) => {
                    if (k > 0) { y += rowH; svg += right(String(s.cur), s.end, y, 'ld-cur'); }
                    if (s.q > 0) {
                        y += rowH;
                        const len = String(s.prod).length;
                        svg += `<text x="${cx(s.end - len) + 6}" y="${y - 4}" class="ld-minus">−</text>` + right(String(s.prod), s.end, y, 'ld-prod');
                        svg += `<line x1="${cx(s.end - len + 1) - w / 2}" y1="${y + 6}" x2="${cx(s.end) + w / 2}" y2="${y + 6}" class="ld-line"/>`;
                    }
                    if (k === steps.length - 1) { y += rowH; svg += right(String(s.rem), s.end, y, 'ld-cur'); }
                });
                document.getElementById('ld-scene').innerHTML = svg;
                const box = document.getElementById('ld-steps');
                box.innerHTML = '';
                shown.forEach((s, k) => { const div = document.createElement('div'); div.textContent = stepText(s, k, steps, n, d); box.appendChild(div); });
                if (shownSteps > steps.length) {
                    const q = n / d, div = document.createElement('div');
                    div.textContent = `Відповідь: ${fmt(n)} : ${d} = ${q}. Перевірка: ${q} · ${d} = ${fmt(n)} ✓`;
                    box.appendChild(div);
                }
                if (!shownSteps) box.textContent = `Скільки буде ${fmt(n)} : ${d}? Спершу знайдіть перше неповне ділене.`;
            };
            window.ldStep = function () {
                const [n, d] = PAIRS[pairIndex];
                shownSteps = Math.min(shownSteps + 1, divSteps(n, d).length + 1);
                renderLongDivision();
            };
            window.ldReset = function () { shownSteps = 0; renderLongDivision(); };
            window.ldPair = function () { pairIndex = (pairIndex + 1) % PAIRS.length; shownSteps = 0; renderLongDivision(); };
