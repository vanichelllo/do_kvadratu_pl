            // --- Слайд 2: прямокутник на клітинках (1 см = 1 клітинка = 40 px), діагоналі ---
            const CELL = 40, OX = 50, OY = 20;
            function rectVals() {
                return [parseInt(document.getElementById('w-a').value, 10) || 1, parseInt(document.getElementById('w-b').value, 10) || 1];
            }
            function rectSvg(a, b) {
                let svg = '';
                for (let x = 0; x <= 8; x++) svg += `<line x1="${OX + x * CELL}" y1="${OY}" x2="${OX + x * CELL}" y2="${OY + 5 * CELL}" class="grid-line"/>`;
                for (let y = 0; y <= 5; y++) svg += `<line x1="${OX}" y1="${OY + y * CELL}" x2="${OX + 8 * CELL}" y2="${OY + y * CELL}" class="grid-line"/>`;
                const x0 = OX, y1 = OY + 5 * CELL, x1 = OX + a * CELL, y0 = y1 - b * CELL;
                svg += `<rect x="${x0}" y="${y0}" width="${a * CELL}" height="${b * CELL}" class="rect${a === b ? ' square' : ''}"/>`
                     + `<line x1="${x0}" y1="${y1}" x2="${x1}" y2="${y0}" class="diag"/><line x1="${x0}" y1="${y0}" x2="${x1}" y2="${y1}" class="diag"/>`
                     + `<text x="${x0 - 12}" y="${y1 + 8}" class="pt-name">A</text><text x="${x0 - 12}" y="${y0 + 4}" class="pt-name">B</text>`
                     + `<text x="${x1 + 12}" y="${y0 + 4}" class="pt-name">C</text><text x="${x1 + 12}" y="${y1 + 8}" class="pt-name">D</text>`;
                return svg;
            }
            window.renderRect = function () {
                const [a, b] = rectVals();
                document.getElementById('w-a-val').textContent = a + ' см';
                document.getElementById('w-b-val').textContent = b + ' см';
                document.getElementById('rect-answer').classList.remove('shown');
                document.getElementById('rect-scene').innerHTML = rectSvg(a, b);
            };
            // Анімація: довжина крок за кроком наближається до ширини — прямокутник стає квадратом
            let sqTimer = null;
            window.makeSquare = function () {
                clearInterval(sqTimer);
                const slider = document.getElementById('w-a'), target = rectVals()[1];
                const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reduce) { slider.value = target; renderRect(); revealRect(); return; }
                sqTimer = setInterval(() => {
                    const a = parseInt(slider.value, 10);
                    if (a === target) { clearInterval(sqTimer); revealRect(); return; }
                    slider.value = a + (a < target ? 1 : -1);
                    renderRect();
                }, 350);
            };
            window.revealRect = function () {
                const [a, b] = rectVals();
                document.getElementById('rect-text').textContent = a === b
                    ? `Квадрат ${a} × ${a} см: усі кути прямі й усі сторони рівні`
                    : `Прямокутник ${a} × ${b} см: усі кути прямі, AB = CD = ${b} см, BC = AD = ${a} см`;
                document.getElementById('rect-answer').classList.add('shown');
            };

            // --- Слайд 4: сторони прямокутника ---
            const STEPS = [
                ['$AB = 7$ см, $BC = 7 - 3 = $ [[$4$ см]].', '«На $3$ см коротша» — віднімаємо'],
                ['Протилежні сторони рівні: $CD = AB = $ [[$7$ см]].', 'Властивість прямокутника'],
                ['І $AD = BC = $ [[$4$ см]].', 'Ще одна пара протилежних сторін'],
                ['Відповідь: $AB = CD = 7$ см, $BC = AD = 4$ см.', 'Готово!'],
            ];
