            window.renderMap = function () {
                let svg = '';
                GROUPS.forEach((name, g) => {
                    const x = 4 + g * GW, cnt = TOPICS.filter(t => t.g === g).length;
                    svg += `<rect x="${x}" y="10" width="${GW - 8}" height="${38 + cnt * 50}" rx="12" class="map-group g${g}"/><text x="${x + 10}" y="28" class="map-group-name g${g}">${name}</text>`;
                });
                TOPICS.forEach((tp, k) => {
                    const x = 4 + tp.g * GW + 8, idx = TOPICS.filter(t => t.g === tp.g).indexOf(tp), y = 40 + idx * 50;
                    svg += `<g class="map-node${mapOn === k ? ' on' : ''}" onclick="pickTopic(${k})"><rect x="${x}" y="${y}" width="${GW - 24}" height="40" rx="8"/>`
                         + `<text x="${x + (GW - 24) / 2}" y="${y + 25}">${tp.n} ${tp.t}</text></g>`;
                });
                document.getElementById('map-scene').innerHTML = svg;
            };
            window.pickTopic = function (k) {
                mapOn = k;

                el.innerHTML = `<b>${tp.n} ${tp.t}.</b> ${tp.d}`;
