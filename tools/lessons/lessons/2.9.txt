            // Паралелепіпед у косій проекції: a — довжина (вправо), b — ширина (углиб), c — висота (вгору)
            function boxPts(x0, y0, a, b, c, s) {
                const dx = b * s * 0.45, dy = -b * s * 0.45;
                const A = [x0, y0], B = [x0 + a * s, y0], C = [x0 + a * s, y0 - c * s], D = [x0, y0 - c * s];
                const sh = (p) => [p[0] + dx, p[1] + dy];
                return { A, B, C, D, A1: sh(A), B1: sh(B), C1: sh(C), D1: sh(D) };
            }
            const ln = (p, q, cls) => `<line x1="${p[0].toFixed(1)}" y1="${p[1].toFixed(1)}" x2="${q[0].toFixed(1)}" y2="${q[1].toFixed(1)}" class="edge${cls ? ' ' + cls : ''}"/>`;
            const poly = (ps, cls) => `<polygon points="${ps.map(p => p[0].toFixed(1) + ',' + p[1].toFixed(1)).join(' ')}" class="face${cls ? ' ' + cls : ''}"/>`;
            // Ребра з групою вимірів: a — довжина, b — ширина, c — висота; hidden — невидиме
            function boxEdges(P) {
                return [
                    [P.A, P.B, 'a', 0], [P.D, P.C, 'a', 0], [P.D1, P.C1, 'a', 0], [P.A1, P.B1, 'a', 1],
                    [P.A, P.D, 'c', 0], [P.B, P.C, 'c', 0], [P.B1, P.C1, 'c', 0], [P.A1, P.D1, 'c', 1],
                    [P.B, P.B1, 'b', 0], [P.C, P.C1, 'b', 0], [P.D, P.D1, 'b', 0], [P.A, P.A1, 'b', 1],
                ];
            }
            function boxSvg(x0, y0, a, b, c, s, mode) {
                const P = boxPts(x0, y0, a, b, c, s);
                let svg = poly([P.A, P.B, P.C, P.D], mode === 'f' ? 'f1' : '')
                    + poly([P.D, P.C, P.C1, P.D1], mode === 'f' ? 'f2' : '')
                    + poly([P.B, P.B1, P.C1, P.C], mode === 'f' ? 'f3' : '');
                boxEdges(P).forEach(([p, q, g, h]) => { svg += ln(p, q, (h ? 'hidden' : '') + (mode === 'e' ? ' g' + g : '')); });
                if (mode === 'v') ['A', 'B', 'C', 'D', 'A1', 'B1', 'C1', 'D1'].forEach((k, i) => {
                    const p = P[k];
                    svg += `<circle cx="${p[0].toFixed(1)}" cy="${p[1].toFixed(1)}" r="7" class="vert"/><text x="${(p[0] + (k.startsWith('A') || k.startsWith('D') ? -14 : 14)).toFixed(1)}" y="${(p[1] + 5).toFixed(1)}" class="vnum">${i + 1}</text>`;
                });
                return svg;
            }
