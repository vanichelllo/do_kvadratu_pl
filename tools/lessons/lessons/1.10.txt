                // Яблука пірамідкою на шальці

                    const row = Math.floor(k / 5), col = k % 5;
                    svg += `<circle cx="${cx - 28 + col * 14 + (row % 2) * 7}" cy="${baseY - 9 - row * 13}" r="7" class="apple${cls}"/>`;
                }
                return svg;
            }
