            // Квадрат 10 × 10 зі стороною s: full — увесь зафарбований (ціла), інакше t стовпчиків і ще h клітинок
            function gridSvg(x, y, s, full, t, h) {
                const c = s / 10;
                let svg = '';
                for (let col = 0; col < 10; col++) for (let row = 0; row < 10; row++) {
                    const cls = full ? ' w' : (col < t ? ' t' : (col === t && row < h ? ' h' : ''));
                    svg += `<rect x="${(x + col * c).toFixed(1)}" y="${(y + row * c).toFixed(1)}" width="${c.toFixed(1)}" height="${c.toFixed(1)}" class="g-cell${cls}"/>`;
                }
                return svg + `<rect x="${x}" y="${y}" width="${s}" height="${s}" class="g-frame"/>`;
            }
            // Смужка з n частин, перші m зафарбовано
            function stripSvg(x, y, w, hgt, n, m, cls) {
                let svg = '';
                for (let k = 0; k < n; k++) svg += `<rect x="${(x + k * w / n).toFixed(1)}" y="${y}" width="${(w / n).toFixed(1)}" height="${hgt}" class="g-cell${k < m ? ' ' + cls : ''}" style="stroke:var(--text-dark);stroke-width:1.5"/>`;
                return svg;
            }

            function gridScene(w, t, h, show) {
                const s = 120, gap = 14, n = w + 1, x0 = (400 - n * s - (n - 1) * gap) / 2;
                let svg = '';
                for (let i = 0; i < w; i++) svg += gridSvg(x0 + i * (s + gap), 30, s, true, 0, 0);
                svg += gridSvg(x0 + w * (s + gap), 30, s, false, t, h);
                svg += lbl(200, 186, show ? `${w},${t}${h}` : 'Яке це число?', show ? 'var(--brand-green)' : 'currentColor');
                return svg;
            }
