            const f1 = (v) => v.toFixed(1);
            // --- Колесо: t від 0 до 1 — частка оберту ---
            window.renderWheel = function (t) {
                const r = 32, L = 2 * Math.PI * r, x0 = 50, y = 170, cx = x0 + L * t, ang = t * 2 * Math.PI;
                let s = `<line x1="20" y1="${y + r}" x2="410" y2="${y + r}" stroke="currentColor" stroke-width="1.5"/>`;
                s += `<line x1="${x0}" y1="${y + r}" x2="${f1(cx)}" y2="${y + r}" class="unroll"/>`;
                s += `<circle cx="${f1(cx)}" cy="${y}" r="${r}" class="circ"/>`;
                // точка обода, що спершу торкалася землі
                const px = cx - r * Math.sin(ang), py = y + r * Math.cos(ang);
                s += `<circle cx="${f1(px)}" cy="${f1(py)}" r="5" fill="var(--warning-red)"/>`;
                if (t >= 1) s += `<text x="${f1(x0 + L / 2)}" y="${y + r + 22}" class="svg-caption">C = πd</text>`;
                document.getElementById('wheel-scene').innerHTML = s;
            };
            window.rollWheel = function () {
                const reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
                const t0 = performance.now(), dur = reduce ? 0 : 2000;
                const step = (now) => {
                    const t = dur ? Math.min((now - t0) / dur, 1) : 1;
                    renderWheel(t);
                    if (t < 1) requestAnimationFrame(step);
                    else document.getElementById('answer-1').classList.add('shown');
                };
                requestAnimationFrame(step);
            };
            // --- Вписані многокутники ---
            window.renderPoly = function () {
                const n = +document.getElementById('w-n').value, R = 100, c = [150, 120];
                document.getElementById('w-n-val').textContent = n;
                const pts = Array.from({ length: n }, (_, k) => [c[0] + R * Math.cos(2 * Math.PI * k / n - Math.PI / 2), c[1] + R * Math.sin(2 * Math.PI * k / n - Math.PI / 2)]);
                document.getElementById('poly-scene').innerHTML = `<circle cx="${c[0]}" cy="${c[1]}" r="${R}" class="circ"/><polygon points="${pts.map(p => p.map(f1).join(',')).join(' ')}" class="poly"/>`;
                const P = n * 2 * Math.sin(Math.PI / n), o = document.getElementById('poly-out');
                o.innerHTML = `$\\dfrac{P_{${n}}}{2R} \\approx ${(P / 2).toFixed(4).replace('.', '{,}')}$ → $\\pi = 3{,}1416\\ldots$`;
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([o]);
            };
            // --- Дуга ---
            window.renderArc = function () {
                const a = +document.getElementById('w-a').value, R = 100, c = [150, 130];
                document.getElementById('w-a-val').textContent = a + '°';
                const s0 = [c[0] + R, c[1]], e = [c[0] + R * Math.cos(-a * Math.PI / 180), c[1] + R * Math.sin(-a * Math.PI / 180)];
                let s = `<circle cx="${c[0]}" cy="${c[1]}" r="${R}" class="circ"/>`;
                s += `<path d="M${f1(s0[0])} ${f1(s0[1])} A${R} ${R} 0 ${a > 180 ? 1 : 0} 0 ${f1(e[0])} ${f1(e[1])}" class="arc-hl"/>`;
                s += `<line x1="${c[0]}" y1="${c[1]}" x2="${f1(s0[0])}" y2="${f1(s0[1])}" class="geo-line sec"/><line x1="${c[0]}" y1="${c[1]}" x2="${f1(e[0])}" y2="${f1(e[1])}" class="geo-line sec"/>`;
                document.getElementById('arc-scene').innerHTML = s;
                document.getElementById('arc-out').classList.remove('shown');
            };
            window.revealArc = function () {
                const a = +document.getElementById('w-a').value, o = document.getElementById('arc-out'), g = (x, y) => y ? g(y, x % y) : x;
                const num = 6 * a, k = g(num, 180), p = num / k, q = 180 / k;
                o.innerHTML = `$l = \\dfrac{\\pi \\cdot 6 \\cdot ${a}}{180} = ${q === 1 ? (p === 1 ? '' : p) + '\\pi' : '\\dfrac{' + p + '\\pi}{' + q + '}'}$`;
                o.classList.add('shown');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([o]);
            };
