                    <p>$\dfrac{1}{a^{-n}} = a^n$: $\dfrac{1}{5^{-2}} = 25$</p>

                    + '</tr><tr><th>$2^n$</th>' + POW.map((p, k) => `<td class="${k >= 3 ? 'new' : ''}">${k < powShown ? '$' + p[1] + '$' : '?'}</td>`).join('') + '</tr></table>'
                    + (powShown === POW.length ? '<p class="center">$2^{-n} = \\dfrac{1}{2^n}$, а $2^0 = 1$</p>' : '');
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([box]);
            };
            window.revealPow = function () { powShown = Math.min(powShown + 1, POW.length); renderPow(); };
