            // Після будь-якої відповіді — позначаємо правильний варіант і показуємо повне розв'язання
            window.checkAnswer = function (quizNum, element, isCorrect) {
                const options = element.parentElement.querySelectorAll('.quiz-option');
                options.forEach(o => {
                    o.classList.remove('correct', 'wrong');
                    if (o.getAttribute('onclick').includes('true')) o.classList.add('correct');
                });
                if (!isCorrect) element.classList.add('wrong');
                const verdict = document.getElementById(`quiz-${quizNum}-verdict`);
                if (verdict) verdict.textContent = isCorrect ? 'Правильно! ✅' : 'Не зовсім — подивись розв\'язання 👇';
                const el = document.getElementById(`quiz-${quizNum}-explanation`);
                if (el) el.style.display = 'block';
                if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise();
            };
